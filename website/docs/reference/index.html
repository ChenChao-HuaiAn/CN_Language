类 工程师 {
私有:
    字符串 名字;
公开:
    函数 工程师(字符串 名) { 名字 = 名; }
    函数 报告(整64 完成数) -> 空类型 {
        打印(名字 + " 完成 " + 整数转字符串(完成数) + " 项");
    }
}

函数 主() -> 整32 {
    向量<任务> 板 = 向量<任务>();

    张三.报告(最大<整64>(板.大小(), 1));
    返回 0;
