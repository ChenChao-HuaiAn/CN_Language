// 递归求和（Task 2.2 递归）：计算数组前 n 个学生的分数总和
不安全 函数 递归分数和(学生* 名单, 整32 n) -> 浮64 {
    如果 (n <= 0) {
        返回 0.0;
    }
    返回 递归分数和(名单, n - 1) + (名单 + (n - 1)).分数;
}

// 计算数组平均分（用递归求总和 + 循环演示）
函数 计算平均分(学生* 名单, 整32 人数) -> 浮64 {
    返回 递归分数和(名单, 人数) / 整32转浮64(人数);
}

// 整32 -> 浮64 显式转换辅助（演示类型转换链）
函数 整32转浮64(整32 v) -> 浮64 {
    返回 v * 1.0;
}

// 根据分数设置等级（选择语句 + 枚举，Task 2.7）
不安全 函数 设置等级(学生* s) {
    浮64 分数 = s.分数;
    如果 (分数 >= 90.0) {
        s.级别 = 等级.优秀;
    } 否则 如果 (分数 >= 75.0) {
        s.级别 = 等级.良好;
    } 否则 如果 (分数 >= 60.0) {
        s.级别 = 等级.及格;
    } 否则 {
        s.级别 = 等级.不及格;
    }
}

// 打印单个学生信息（-> 成员访问 + 枚举case，Task 2.7）

    打印("  分数: ", s.分数);

        学生{ 名字 = "李四", 分数 = 78.0, 级别 = 等级.良好 },
        学生{ 名字 = "王五", 分数 = 55.5, 级别 = 等级.不及格 },
        学生{ 名字 = "赵六", 分数 = 66.5, 级别 = 等级.及格 },
        学生{ 名字 = "钱七", 分数 = 88.0, 级别 = 等级.良好 }
    };
    整32 人数 = 5;

    // ---- 2. 循环遍历 + 指针算术（Task 2.4 指针遍历 + Task 2.1 循环）----
    打印("---- 原始名单 ----");
    循环 (整32 i = 0; i < 人数; i++) {
        打印学生(&名单[i]);
    }

    // ---- 3. 数组名退化 + 指针遍历 + 继续（Task 2.4 + Task 2.1）----
    打印("---- 指针遍历（跳过索引2）----");
    学生* p = 名单;
    整32 索引 = 0;
    当 (p < 名单 + 人数) {
        如果 (索引 == 2) {
            索引++;
            p++;
            继续;
        }
        打印("  第", 索引, "个: ", p.名字);
        索引++;
        p++;
    }

    // ---- 4. 递归求平均分（Task 2.2 递归 + Task 2.3 浮点运算）----
    打印("---- 统计 ----");
    打印("总人数: ", 人数);
    浮64 平均分 = 计算平均分(名单, 人数);
    打印("平均分: ", 平均分);

    // ---- 5. 函数指针回调做排序（Task 2.2 函数指针，定义见文件末尾）----
    // 说明：结构体整体赋值/拷贝未实现（阶段2范围外），排序采用"索引数组"：
    //       交换索引而非结构体元素，函数指针比较 名单[顺序[i]] 的分数
    打印("---- 按分数排序（索引数组）----");
    整32[5] 顺序 = { 0, 1, 2, 3, 4 };
    整32(*比较)(学生*, 学生*) = 按分数比较;
    索引排序(名单, 顺序, 人数, 比较);
    循环 (整32 j = 0; j < 人数; j++) {
        打印学生(&名单[顺序[j]]);

    打印("---- 字符串 API ----");

    打印("标题长度: ", 字符串长度(报告头));

    字符串 第一行 = 名单[0].名字 + " 是第一名";
    打印("连接: ", 第一行);

    // ---- 7. 原始/多行字符串（Task 2.5）----
    打印(原始"原始串: C:\dir\file");
    打印(多行"""多行: 第1行\n第2行""");

    // ---- 8. 位运算/移位/后缀字面量/隐式转换（Task 2.3）----
    打印("---- 类型系统 ----");
    整32 掩码 = 0b1100;

    打印("位与: ", 掩码 & 数据);
    打印("位或: ", 掩码 | 数据);

    打印("左移: ", 掩码 << 2);
    打印("右移: ", 掩码 >> 1);

    正32 无符号 = 4000000000U;
    打印("后缀L: ", 大值);
    打印("后缀U: ", 无符号);
    浮32 单精度 = 3.5f;
    浮64 双精度 = 单精度;
    打印("浮32->浮64: ", 双精度);
    整8 小值 = 100;
    整16 中值 = 30000;
    打印("整8+50: ", 小值 + 50);

// 索引排序（冒泡思想，交换索引而非结构体元素）：函数指针作为回调参数（Task 2.2）
// 结构体整体赋值/拷贝未实现（阶段2范围外），故只交换 顺序 数组里的索引值
不安全 函数 索引排序(学生* 名单, 整32* 顺序, 整32 人数, 整32(*比较)(学生*, 学生*)) {
    循环 (整32 i = 0; i < 人数 - 1; i++) {
        循环 (整32 j = 0; j < 人数 - 1 - i; j++) {
            如果 (比较(&名单[顺序[j]], &名单[顺序[j + 1]]) > 0) {
                整32 临时索引 = 顺序[j];
                顺序[j] = 顺序[j + 1];
                顺序[j + 1] = 临时索引;
            }
        }
    }
}

// 按分数降序比较（分数高者排前）
不安全 函数 按分数比较(学生* a, 学生* b) -> 整32 {
    如果 (a.分数 < b.分数) {
        返回 1;
    }
    如果 (a.分数 > b.分数) {
        返回 -1;
    }
    返回 0;
}
