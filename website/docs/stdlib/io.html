IO

标准输入与错误输出——四个函数，全部返回 结果 以显式处理失败。

导入写法CN

导入 IO::{读取行, 读取整数, 读取浮点, 打印到错误};

标准输入

EOF（无数据）或非法输入时返回 错误(错误码.文件)，不做静默垃圾值。读取行 返回拥有型字符串（RAII 自动释放）。

标准输入.cnCN

导入 IO::{读取行, 读取整数};

函数 主() -> 整32 {
    打印("名字？");
    结果<字符串, 整32> 名字 = 读取行();
    如果 (名字.正常) {
        打印("你好，", 名字.值);
    } 否则 {
        打印("(输入已关闭)");        // EOF 分支
    }
    结果<整64, 整32> 年龄 = 读取整数();
    如果 (年龄.正常) {
        打印("明年 ", 年龄.值 + 1, " 岁");
    } 否则 {
        打印("(没有读到整数)");
    }
    返回 0;
}

输出：名字？ (输入已关闭) (没有读到整数)

读取行

函数 读取行() -> 结果<字符串, 整32>

从标准输入读一行（不含行尾换行符）。EOF 时返回错误。返回字符串为拥有型，作用域结束自动释放。

返回　正常(一行文本)；EOF → 错误(错误码.文件)。

主.cnCN

函数 主() -> 整32 {
    打印("===== IO 输入库（Task 6.2）=====");

    // ---- 1. 读取行（整行字符串，不含换行）----
    字符串 行 = IO::读取行();
    如果 (行 != 无) {
        打印("读取行: ", 行);
    } 否则 {
        打印("读取行失败(EOF)");
    }

    // ---- 2. 读取整数（strtoll 整行解析，成功标志输出）----
    整32 整数标志 = 0;
    整64 整数值 = IO::读取整数(&整数标志);
    如果 (整数标志 == 1) {
        打印("读取整数: ", 整数值);
    } 否则 {
        打印("读取整数失败");
    }

    // ---- 3. 读取浮点（strtod 整行解析，成功标志输出）----
    整32 浮点标志 = 0;
    浮64 浮点值 = IO::读取浮点(&浮点标志);
    如果 (浮点标志 == 1) {
        打印("读取浮点: ", 浮点值);
    } 否则 {
        打印("读取浮点失败");
    }

    // ---- 4. EOF 语义：输入已耗尽，读取整数 触发 EOF 返回成功标志0 ----
    整32 结束标志 = 1;
    整64 结束值 = IO::读取整数(&结束标志);
    如果 (结束标志 == 1) {
        打印("EOF读取整数: ", 结束值);
    } 否则 {
        打印("EOF读取整数失败");
    }

    打印("===== IO 输入库测试完成 =====");
    返回 0;
}

// ==================== 统计计算（算法.cn 排序/最大元素 + 数学.cn 平方根） ====================
不安全 函数 统计计算(整64* 总分数组, 整64 总人数) -> 统计结果 {
    排序整64(总分数组, 总人数);
    整64 总分和 = 0;
    整64 累加索引 = 0;
    当 (累加索引 < 总人数) {
        总分和 = 总分和 + 总分数组[累加索引];
        累加索引++;
    }
    整64 平均分 = 总分和 / 总人数;
    整64 最高分 = 最大元素<整64>(总分数组, 总人数);
    整64 最低分 = 总分数组[0];
    整64 及格人数 = 0;
    整64 及格索引 = 0;
    当 (及格索引 < 总人数) {
        如果 (总分数组[及格索引] >= 60) {
            及格人数++;
        }
        及格索引++;
    }
    整64 差异和 = 0;
    整64 差索引 = 0;
    当 (差索引 < 总人数) {
        整64 差异 = 总分数组[差索引] - 平均分;
        差异和 = 差异和 + 差异 * 差异;
        差索引++;
    }
    浮64 方差 = 浮64(差异和) / 浮64(总人数);
    浮64 标准差 = 数学::平方根(方差);
    统计结果 统计 = 统计结果{ 平均分 = 平均分, 最高分 = 最高分, 最低分 = 最低分, 及格人数 = 及格人数, 标准差 = 标准差 };
    返回 统计;
}

// ==================== 解析单行（字符串扩展.cn 分割/转整数 + 深拷贝） ====================
// 说明：返回 1=成功写入 0=跳过（非法格式/解析失败）
不安全 函数 解析单行(字符串 行, 整64* 总分数组, 字符串* 姓名数组, 整64 位置) -> 整64 {
    字符串[4] 片段;
    结果<整64, 整32> 分割结果 = 分割(行, ",", 片段, 4);
    整64 成功 = 0;
    如果 (分割结果.正常) {
        如果 (分割结果.值 == 4) {
            整32 标志1 = 0;
            整32 标志2 = 0;
            整32 标志3 = 0;
            整64 成绩一 = 解析::转整数(片段[1], &标志1);
            整64 成绩二 = 解析::转整数(片段[2], &标志2);
            整64 成绩三 = 解析::转整数(片段[3], &标志3);
            如果 (标志1 == 1 && 标志2 == 1 && 标志3 == 1) {
                // 姓名深拷贝（分割产物释放后不得悬垂）；总分 = 三科之和
                字符串 姓名副本 = 字符串复制(片段[0]);
                整64 总分 = 成绩一 + 成绩二 + 成绩三;
                总分数组[位置] = 总分;
                姓名数组[位置] = 姓名副本;
                成功 = 1;
            }
        }
    }
    整64 释放索引 = 0;
    当 (释放索引 < 4) {
        字符串释放(片段[释放索引]);
        释放索引++;
    }
    返回 成功;
}

// ==================== 解析数据文件（文件.cn + 解析单行） ====================
// 返回 有效总人数；打开失败 返回 0
函数 解析数据(字符串 数据路径, 整64* 总分数组, 字符串* 姓名数组, 整64 最大容量) -> 整64 {
    空类型* 读句柄 = 文件::打开文件(数据路径, 1);
    如果 (读句柄 == 无) {
        IO::打印到错误("打开数据文件失败，错误码=5(文件)\n");
        返回 0;
    }
    整64 跳过行数 = 0;
    整64 总人数 = 0;
    当 (真) {
        字符串 行 = 文件::读取文件行(读句柄);
        如果 (行 == 无) {
            中断;    // EOF
        }
        如果 (总人数 < 最大容量) {
            整64 成功 = 解析单行(行, 总分数组, 姓名数组, 总人数);
            如果 (成功 == 1) {
                总人数++;
            } 否则 {
                跳过行数++;
            }
        } 否则 {
            跳过行数++;
        }
    }
    文件::关闭文件(读句柄);
    打印("跳过非法行: ", 跳过行数);
    返回 总人数;
}

// ==================== 释放姓名数组 ====================
不安全 函数 释放姓名(字符串* 姓名数组, 整64 总人数) -> 空类型 {
    整64 释放姓名 = 0;
    当 (释放姓名 < 总人数) {
        字符串释放(姓名数组[释放姓名]);
        释放姓名++;
    }
}
