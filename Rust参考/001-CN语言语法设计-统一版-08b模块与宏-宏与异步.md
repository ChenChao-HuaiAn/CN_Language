## 四、宏系统

### 4.1 声明宏

```ebnf
宏定义 = "宏" 标识符 "!" "(" 宏参数? ")" "{" 宏规则+ "}"
宏规则 = 模式 "=>" 转换
宏参数 = "$" 标识符 ":" 片段说明符
片段说明符 = "标识" | "表达式" | "类型" | "模式" | "语句" | "块" | "路径"
```

```cn
// 基本声明宏
宏 说你好!() {
    打印行!("你好！");
}

说你好!();                       // 调用宏

// 带参数的声明宏
宏 创建函数!($名称:标识, $值:表达式) {
    函数 $名称() -> 整32 { $值 }
}

创建函数!(获取零, 0);
创建函数!(获取一, 1);
// 等价于定义了 函数 获取零() -> 整32 { 0 } 和 函数 获取一() -> 整32 { 1 }

// 宏的模式匹配
宏 匹配值!($值:表达式 { $($模式:模式 => $结果:表达式),* }) {
    匹配 $值 {
        $($模式 => $结果),*
    }
}
```

### 4.2 过程宏

#### 派生宏

```cn
// 定义派生宏（在单独的宏包中）
#[过程宏_派生]
公开 函数 生成问候(项: 语法树) -> 令牌流 {
    // 解析输入语法树，生成实现代码
}

// 使用派生宏
#[派生(问候)]
结构 用户 {
    字符串 姓名,
}
```

#### 属性宏

```cn
// 定义属性宏
#[过程宏_属性]
公开 函数 路由(属性: 令牌流, 项: 令牌流) -> 令牌流 {
    // 处理属性和被标注的项
}

// 使用属性宏
#[路由("/用户/列表")]
异步 函数 用户列表() -> 响应 {
    // ...
}
```

#### 函数式宏

```cn
// 定义函数式宏
#[过程宏_函数]
公开 函数 查询(输入: 令牌流) -> 令牌流 {
    // 将SQL查询语法转换为代码
}

// 使用
让 结果 = 查询!("SELECT * FROM users WHERE age > 18");
```

## 五、异步编程

### 5.1 异步函数

```ebnf
异步函数 = "异步" "函数" 标识符 "(" 参数列表? ")" ("->" 类型)? 块表达式
```

```cn
异步 函数 获取数据(&文 网址) -> 结果<字符串, 网络错误> {
    让 响应 = 网络::异步获取(网址).等待?;
    让 内容 = 响应.读取文本().等待?;
    返回 正常(内容);
}
```

### 5.2 异步块

```ebnf
异步块 = "异步" 块表达式
```

```cn
让 未来 = 异步 {
    让 数据 = 读取文件("数据.txt").等待?;
    让 解析 = 解析数据(&数据).等待?;
    正常(解析)
};
```

### 5.3 生成任务

```cn
异步 函数 主() {
    让 任务1 = 生成 异步 获取数据("https://示例.cn/数据1");
    让 任务2 = 生成 异步 获取数据("https://示例.cn/数据2");

    // 并发等待
    让 (结果1, 结果2) = 加入!(任务1, 任务2).等待;

    打印行!("结果1: {:?}", 结果1);
    打印行!("结果2: {:?}", 结果2);
}
```

## 六、不安全代码

### 6.1 不安全块

```ebnf
不安全块 = "不安全" 块表达式
```

```cn
不安全 {
    // 可以执行以下操作：
    // 1. 解引用裸指针
    // 2. 调用不安全函数
    // 3. 访问或修改可变静态变量
    // 4. 实现不安全特征
}
```

### 6.2 不安全函数

```ebnf
不安全函数 = "不安全" "函数" 标识符 "(" 参数列表? ")" ("->" 类型)? 块表达式
```

```cn
不安全 函数 读取裸指针(*常量 整32 指针) -> 整32 {
    *指针                         // 解引用裸指针
}
```

### 6.3 不安全实现

```cn
不安全 实现 发送 对于 我的类型 {}    // 手动承诺线程安全
```

> **设计决策**：规范总结使用指针类型（整数*）和指针运算，统一后采用所有权+借用系统管理内存，裸指针操作限制在不安全块内。不安全是Rust的unsafe的中文翻译，表示"编译器无法验证安全性的代码"。

## 七、FFI外部函数（v1.2.0-alpha.1新增）

### 7.1 extern块

```ebnf
extern块 = "外部" 字符串字面量 "{" extern函数声明* "}"
extern函数声明 = "不安全"? "函数" 标识符 "(" extern参数列表? ")" ("->" 类型)? ";"
extern参数列表 = extern参数 ("," extern参数)* ("," "...")?
extern参数 = 标识符 ":" 类型 | 类型
```

```cn
外部 "C" {
    函数 abs(值: 整32) -> 整32;                 // C标准库abs
    函数 malloc(大小: 整64) -> *可变 整8;        // C标准库malloc
    函数 free(指针: *可变 整8);                  // 无返回类型=void
    不安全 函数 printf(格式: *只读 整8, ...) -> 整32;  // 可变参数varargs
}
```

- **`外部`**：声明外部函数块，`"C"`指定C ABI
- extern块内函数使用**C风格参数语法**：`名称: 类型`（名称在前）
- 参数类型必须是FFI兼容类型（基础类型、裸指针）
- 返回类型可省略（映射C的void），或用`空`显式声明
- `...`表示C可变参数（varargs），如printf
- extern块内函数**默认不安全**，`不安全`关键字可选
- extern函数只有签名没有函数体，以分号结尾

### 7.2 调用外部函数

外部函数只能在**不安全块**或**不安全函数**中调用：

```cn
函数 主() -> 整32 {
    让 结果 = 不安全 { abs(-42) };        // 不安全块中调用
    打印整数(结果);                       // 输出42
    返回 0;
}
```

### 7.3 ABI支持

| ABI名称 | 说明 |
|---------|------|
| `"C"` | C ABI（默认，v1.2.0-alpha.1支持） |
| `"stdcall"` | Windows stdcall（预留） |
| `"fastcall"` | 快速调用约定（预留） |

> **设计决策**：`外部`对应Rust的extern关键字，`"C"`字符串指定ABI。extern块内函数默认不安全（FFI调用编译器无法验证安全性），与Rust一致。参数语法采用C风格`名称: 类型`以贴近C声明习惯，与CN普通函数的类型前置语法（`函数 f(类型 名称)`）不同，两者在extern边界自然区分。
