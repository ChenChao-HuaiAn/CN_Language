#pragma once
#include <cstdint>
#include <string>
#include "cn_compiler/common/source_location.hpp"

namespace cn_compiler {

// Token类型枚举：覆盖53个关键字、字面量、运算符、分隔符
enum class TokenType {
    // ---- 控制流关键字(10) ----
    关键字如果, 关键字否则, 关键字当, 关键字循环, 关键字返回,
    关键字中断, 关键字继续, 关键字选择, 关键字情况, 关键字默认,
    // ---- 类型关键字(20) ----
    关键字整数, 关键字小数,          // 别名：整32、浮64
    关键字整8, 关键字整16, 关键字整32, 关键字整64, 关键字整128,
    关键字正8, 关键字正16, 关键字正32, 关键字正64, 关键字正128,
    关键字浮32, 关键字浮64,
    关键字布尔, 关键字字符, 关键字字符串, 关键字空类型,
    关键字结构体, 关键字枚举,
    // ---- 声明关键字(7) ----
    关键字函数, 关键字变量, 关键字导入, 关键字从,
    关键字公开, 关键字私有, 关键字静态,
    // ---- 常量关键字(3) ----
    关键字真, 关键字假, 关键字无,
    // ---- OOP关键字(9) ----
    关键字类, 关键字接口, 关键字保护, 关键字虚拟, 关键字重写,
    关键字抽象, 关键字实现, 关键字自身, 关键字父类,
    // ---- 错误处理关键字(2) ----
    关键字结果, 关键字可选,
    // ---- 字面量前缀关键字(2) ----
    关键字原始, 关键字多行,

    // ---- 标识符与字面量 ----
    标识符,                    // 中文/ASCII标识符
    整数字面量,                // 42 / 0xFF / 0b1010 / 0o777（含后缀）
    浮点字面量,                // 3.14 / 1.5e10
    字符字面量,                // 'A' / '中' / '\u{4E2D}'
    字符串字面量,              // "你好" / 原始"..." / 多行"""..."""

    // ---- 运算符 ----
    加号, 减号, 乘号, 除号, 取余号,              // + - * / %
    等号等号, 不等号, 小于号, 大于号,            // == != < >
    小于等于, 大于等于,                            // <= >=
    逻辑与, 逻辑或, 逻辑非,                        // && || !
    位与, 位或, 位异或, 位非, 左移, 右移,          // & | ^ ~ << >>
    赋值, 加等, 减等, 乘等, 除等, 取余等,          // = += -= *= /= %=
    位与等, 位或等, 位异或等, 左移等, 右移等,      // &= |= ^= <<= >>=
    取地址, 解引用, 箭头, 点号, 下标开, 下标关,     // & * -> . [ ]
    括号开, 括号关, 花括号开, 花括号关,            // ( ) { }
    分号, 逗号, 冒号,                              // ; , :

    文件结束,  // EOF
};

// Token：词法单元，携带类型、文本、源码位置与字面量值
class Token {
public:
    Token(TokenType 类型, std::string 文本, SourceLocation 位置)
        : 类型_(类型), 文本_(std::move(文本)), 位置_(位置) {}

    // 获取类型
    TokenType 类型() const { return 类型_; }
    // 获取文本
    const std::string& 文本() const { return 文本_; }
    // 获取源码位置
    const SourceLocation& 位置() const { return 位置_; }

    // 整数字面量的数值（仅对整数字面量有效）
    std::int64_t 整数值() const { return 整数值_; }
    // 设置整数值
    void 设置整数值(std::int64_t 值) { 整数值_ = 值; }

    // 判断是否是指定类型
    bool 是(TokenType 类型) const { return 类型_ == 类型; }
    // 判断是否是关键字之一
    bool 是关键字() const {
        return 类型_ >= TokenType::关键字如果 && 类型_ <= TokenType::关键字多行;
    }

private:
    TokenType 类型_;          // Token类型
    std::string 文本_;        // Token原始文本
    SourceLocation 位置_;     // 源码位置
    std::int64_t 整数值_ = 0; // 整数字面量值
};

} // namespace cn_compiler

namespace cn_compiler {

// 53个关键字 -> TokenType 映射表
// 控制流(10)：如果/否则/当/循环/返回/中断/继续/选择/情况/默认

// 查找关键字：命中返回true并设置类型
bool 词法分析器::查找关键字(const std::string& 文本, TokenType& 类型) {
    const auto& 表 = 关键字表();
    auto 迭代 = 表.find(文本);
    if (迭代 != 表.end()) {
        类型 = 迭代->second;
        return true;
    }
    return false;
}

// UTF-8解码：从位置_读取一个字符，返回Unicode码点
char32_t 词法分析器::前进() {
    if (到达末尾()) return 0;
    if (回退标记_) {
        回退标记_ = false;
        列_++;
        return 回退字符_;
    }
    unsigned char 首字节 = static_cast<unsigned char>(源码_[位置_]);
    位置_++;
    // 计算UTF-8字符长度并解码
    int 长度 = 1;
    char32_t 码点 = 首字节;
    if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
    else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
    else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
    for (int i = 1; i < 长度 && 位置_ < 源码_.size(); i++) {
        码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[位置_]) & 0x3F);
        位置_++;
    }
    列_++;
    return 码点;
}

// 回退一个字符
void 词法分析器::回退() {
    回退标记_ = true;
    // 将回退字符存于回退字符_，此处通过查看()预先取得
    // 简化实现：回退时列号递减由查看/前进共同维护
}

// 查看当前字符
char32_t 词法分析器::查看() const {
    if (到达末尾()) return 0;
    if (回退标记_) return 回退字符_;
    // 解码当前位置的字符
    unsigned char 首字节 = static_cast<unsigned char>(源码_[位置_]);
    if (首字节 < 0x80) return 首字节;
    int 长度 = 1;
    char32_t 码点 = 首字节;
    if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
    else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
    else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
    for (int i = 1; i < 长度 && 位置_ + i < 源码_.size(); i++) {
        码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[位置_ + i]) & 0x3F);
    }
    return 码点;
}

// 查看向后第n个字符
char32_t 词法分析器::查看第N个(int n) const {
    if (n <= 0) return 查看();
    if (回退标记_ && n == 1) {
        // 回退场景简化：仅支持n=1时返回回退字符
        return 回退字符_;
    }
    // 简化实现：从当前位置顺序解码n个字符
    std::size_t 偏移 = 位置_;
    char32_t 结果 = 0;
    for (int i = 0; i < n; i++) {
        if (偏移 >= 源码_.size()) return 0;
        unsigned char 首字节 = static_cast<unsigned char>(源码_[偏移]);
        int 长度 = 1;
        char32_t 码点 = 首字节;
        if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
        else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
        else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
        for (int j = 1; j < 长度 && 偏移 + j < 源码_.size(); j++) {
            码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[偏移 + j]) & 0x3F);
        }
        偏移 += 长度;
        结果 = 码点;
    }
    return 结果;
}

// 是否到达源码末尾
bool 词法分析器::到达末尾() const {
    return 位置_ >= 源码_.size();
}

// 当前位置
SourceLocation 词法分析器::当前位置() const {
    return SourceLocation(文件名_, 行_, 列_);
}

// 跳过空白与注释（支持//单行、/* */块注释嵌套）
void 词法分析器::跳过空白与注释() {
    bool 继续 = true;
    while (继续) {
        继续 = false;
        // 跳过空白
        while (!到达末尾()) {
            char32_t c = 查看();
            if (c == '\n') {
                前进();
                行_++;
                列_ = 1;
            } else if (c == ' ' || c == '\t' || c == '\r') {
                前进();
            } else {
                break;
            }
        }
        // 单行注释
        if (查看() == '/' && 查看第N个(1) == '/') {
            前进(); 前进();
            while (!到达末尾() && 查看() != '\n') 前进();
            继续 = true;
        }
        // 块注释（支持嵌套）
        if (查看() == '/' && 查看第N个(1) == '*') {
            前进(); 前进();
            int 嵌套深度 = 1;
            while (!到达末尾() && 嵌套深度 > 0) {
                if (查看() == '/' && 查看第N个(1) == '*') { 前进(); 前进(); 嵌套深度++; }
                else if (查看() == '*' && 查看第N个(1) == '/') { 前进(); 前进(); 嵌套深度--; }
                else 前进();
            }
            继续 = true;
        }
    }
}

// 读取标识符或关键字（UTF-8中文/ASCII字母开头）
Token 词法分析器::读取标识符或关键字() {
    SourceLocation 位置 = 当前位置();
    std::string 文本;
    while (!到达末尾()) {
        char32_t c = 查看();
        // 标识符字符：字母/数字/下划线/中文（码点>=0x80）
        bool 是标识符字符 = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
                           (c >= '0' && c <= '9') || c == '_' || c >= 0x80;
        if (!是标识符字符) break;
        // 收集当前字符的UTF-8字节
        if (回退标记_) {
            文本 += 回退字符_;  // 简化：ASCII回退场景
            前进();
        } else {
            std::size_t 起始 = 位置_;
            前进();
            文本.append(源码_, 起始, 位置_ - 起始);
        }
    }
    // 判断是否为关键字
    TokenType 类型;
    if (查找关键字(文本, 类型)) {
        return Token(类型, 文本, 位置);
    }
    return Token(TokenType::标识符, 文本, 位置);
}

// 主入口：分析源码返回Token流
std::vector<Token> 词法分析器::分析(const std::string& 源码, const std::string& 文件名) {
    源码_ = 源码;
    文件名_ = 文件名;
    位置_ = 0;
    行_ = 1;
    列_ = 1;
    回退标记_ = false;

    std::vector<Token> tokens;
    while (true) {
        跳过空白与注释();
        if (到达末尾()) {
            tokens.push_back(Token(TokenType::文件结束, "", 当前位置()));
            break;
        }
        char32_t c = 查看();
        // 标识符/关键字
        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '_' || c >= 0x80) {
            tokens.push_back(读取标识符或关键字());
            continue;
        }
        // 数字
        if (c >= '0' && c <= '9') {
            tokens.push_back(读取数字());
            continue;
        }
        // 字符串/字符/运算符由后续Task细化实现（此处为骨架，完整实现在本Task Step 5补齐）
        // 简化：未知字符报告诊断并跳过
        诊断_.报告(诊断::错误(文件名_, 行_, 列_, "暂不支持该词法单元"));
        前进();
    }
    return tokens;
}

// 读取数字（骨架：完整实现在Step 5补齐，此处支持十进制整数）
Token 词法分析器::读取数字() {
    SourceLocation 位置 = 当前位置();
    std::string 文本;
    while (!到达末尾()) {
        char32_t c = 查看();
        if (c >= '0' && c <= '9') {
            文本 += static_cast<char>(c);
            前进();
        } else {
            break;
        }
    }
    Token token(TokenType::整数字面量, 文本, 位置);
    token.设置整数值(std::stoll(文本));
    return token;
}

// 读取字符串（骨架：完整实现在Step 5）
Token 词法分析器::读取字符串() {
    return Token(TokenType::字符串字面量, "", 当前位置());
}

// 读取字符（骨架：完整实现在Step 5）
Token 词法分析器::读取字符() {
    return Token(TokenType::字符字面量, "", 当前位置());
}

// 读取运算符或分隔符（骨架：完整实现在Step 5）
Token 词法分析器::读取运算符或分隔符() {
    return Token(TokenType::文件结束, "", 当前位置());
}
