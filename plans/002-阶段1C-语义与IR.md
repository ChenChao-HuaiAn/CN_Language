// 符号表：作用域链管理，记录变量/函数/类型符号
#pragma once
#include <optional>
#include <string>
#include <unordered_map>
#include <vector>

namespace cn_compiler {

// 符号表：支持嵌套作用域，同一作用域内禁止重复声明
class 符号表 {
public:
    // 进入新作用域
    void 进入作用域() { 作用域栈_.emplace_back(); }
    // 退出当前作用域
    void 退出作用域() { if (!作用域栈_.empty()) 作用域栈_.pop_back(); }

    // 插入符号：同作用域重复返回false
    bool 插入(const std::string& 名称, const std::string& 类型) {
        if (作用域栈_.empty()) 进入作用域();
        auto& 当前 = 作用域栈_.back();
        if (当前.find(名称) != 当前.end()) return false;
        当前[名称] = 类型;
        return true;
    }

    // 查找符号：从内到外，返回类型
    std::optional<std::string> 查找(const std::string& 名称) const {
        for (auto 迭代 = 作用域栈_.rbegin(); 迭代 != 作用域栈_.rend(); ++迭代) {
            auto 找到 = 迭代->find(名称);
            if (找到 != 迭代->end()) return 找到->second;
        }
        return std::nullopt;
    }

private:
    std::vector<std::unordered_map<std::string, std::string>> 作用域栈_;  // 作用域链
};

} // namespace cn_compiler
```

// 类型系统：CN类型表示、大小、隐式转换规则
#pragma once
#include <optional>
#include <string>
#include <unordered_set>

namespace cn_compiler {

// CN类型表示
class CN类型 {
public:
    explicit CN类型(std::string 名称) : 名称_(std::move(名称)) {}
    // 获取类型名
    const std::string& 名称() const { return 名称_; }
    // 是否为空类型
    bool 是空类型() const { return 名称_ == "空类型"; }

private:
    std::string 名称_;
};

// 类型系统：查询类型信息与隐式转换合法性
class 类型系统 {
public:
    // 判断源类型能否隐式转换到目标类型
    // 规则（规格书3.7）：字符↔整数、字符串↔字符*、空类型*↔任意指针、
    // 整数->整64（宽化）、浮32->浮64（宽化）
    static bool 可隐式转换(const std::string& 源, const std::string& 目标) {
        if (源 == 目标) return true;
        // 字符 ↔ 整数
        if ((源 == "字符" && 目标 == "整数") || (源 == "整数" && 目标 == "字符")) return true;
        // 字符串 ↔ 字符*
        if ((源 == "字符串" && 目标 == "字符*") || (源 == "字符*" && 目标 == "字符串")) return true;
        // 整数 -> 整64 宽化
        if ((源 == "整数" && 目标 == "整64") || (源 == "整32" && 目标 == "整64")) return true;
        // 浮32 -> 浮64 宽化
        if (源 == "浮32" && 目标 == "浮64") return true;
        // 空类型* -> 任意指针
        if (源 == "空类型*" && !目标.empty() && 目标.back() == '*') return true;
        return false;
    }

    // 判断是否为内置类型关键字
    static bool 是内置类型(const std::string& 名称) {
        static const std::unordered_set<std::string> 内置类型 = {
            "整数", "小数", "整8", "整16", "整32", "整64", "整128",
            "正8", "正16", "正32", "正64", "正128",
            "浮32", "浮64", "布尔", "字符", "字符串", "空类型",
        };
        return 内置类型.count(名称) > 0;
    }

    // 获取类型大小（字节）
    static int 大小(const std::string& 名称) {
        if (名称 == "整8" || 名称 == "正8" || 名称 == "布尔") return 1;
        if (名称 == "整16" || 名称 == "正16") return 2;
        if (名称 == "整32" || 名称 == "正32" || 名称 == "浮32") return 4;
        if (名称 == "整64" || 名称 == "正64" || 名称 == "浮64" || 名称 == "字符") return 8;
        if (名称 == "整128" || 名称 == "正128") return 16;
        if (名称 == "整数") return 4;   // 整32别名
        if (名称 == "小数") return 8;   // 浮64别名
        return 0;  // 空类型/自定义类型由语义分析器计算
    }
};

} // namespace cn_compiler

#include <memory>
#include "cn_compiler/common/diagnostics.hpp"

#include "cn_compiler/semantic/symbol_table.hpp"

// 语义分析器实现：符号收集 + 基础类型检查（完整类型推断在阶段二完善）
#include "cn_compiler/semantic/semantic.hpp"

namespace cn_compiler {

// 检查程序
bool 语义分析器::检查(std::unique_ptr<程序>& 程序) {
    符号表_.进入作用域();  // 全局作用域
    // 预置内置函数符号（打印行等运行时函数，阶段一先注册原型）
    符号表_.插入("打印行", "空类型");
    // 遍历顶层声明
    for (auto& 声明 : 程序->顶层声明) {
        检查顶层声明(*声明);
    }
    return 诊断_.是否成功();
}

// 检查顶层声明
void 语义分析器::检查顶层声明(AST节点& 声明) {
    if (声明.类型() == 节点类型::函数声明) {
        检查函数(static_cast<函数声明&>(声明));
    }
    // 结构体/枚举/类/导入：阶段二/三实现
}

// 检查函数
void 语义分析器::检查函数(函数声明& 函数) {
    // 注册函数符号（重载支持：阶段二用签名区分）
    if (!符号表_.插入(函数.名称, 函数.返回类型.empty() ? "空类型" : 函数.返回类型)) {
        诊断_.报告(诊断::错误(函数.位置, "重复声明函数 '" + 函数.名称 + "'"));
        return;
    }
    // 无函数体（原型声明）直接返回
    if (!函数.函数体) return;
    // 参数作用域
    符号表_.进入作用域();
    for (auto& 参数 : 函数.参数列表) {
        if (!符号表_.插入(参数->名称, 参数->类型)) {
            诊断_.报告(诊断::错误(参数->位置, "重复声明参数 '" + 参数->名称 + "'"));
        }
    }
    // 检查函数体语句
    for (auto& 语句 : 函数.函数体->语句列表) {
        检查语句(*语句);
    }
    符号表_.退出作用域();
}

// 检查语句（骨架：完整分支在阶段二补齐，阶段一覆盖返回/表达式/变量声明）
void 语义分析器::检查语句(语句& 语句节点) {
    switch (语句节点.类型()) {
        case 节点类型::返回语句: {
            auto& 返回 = static_cast<返回语句&>(语句节点);
            if (返回.值) 检查表达式(*返回.值);
            break;
        }
        case 节点类型::表达式语句: {
            auto& 表达式语句 = static_cast<cn_compiler::表达式语句&>(语句节点);
            检查表达式(*表达式语句.表达式);
            break;
        }
        case 节点类型::变量声明: {
            auto& 声明 = static_cast<变量声明&>(语句节点);
            std::string 类型 = 声明.类型;
            // 类型推断：无显式类型时从初始值推断
            if (类型.empty() && 声明.初始值) {
                类型 = 检查表达式(*声明.初始值);
            } else if (声明.初始值) {
                std::string 初始类型 = 检查表达式(*声明.初始值);
                if (!类型系统::可隐式转换(初始类型, 类型)) {
                    诊断_.报告(诊断::错误(声明.位置,
                        "无法将 '" + 初始类型 + "' 隐式转换为 '" + 类型 + "'"));
                }
            }
            if (!符号表_.插入(声明.名称, 类型)) {
                诊断_.报告(诊断::错误(声明.位置, "重复声明变量 '" + 声明.名称 + "'"));
            }
            break;
        }
        default:
            // 控制流语句：阶段二实现
            break;
    }
}

// 检查表达式（骨架：完整类型推断在阶段二完善）
std::string 语义分析器::检查表达式(表达式& 表达式节点) {
    switch (表达式节点.类型()) {
        case 节点类型::字面量表达式: {
            auto& 字面量 = static_cast<字面量表达式&>(表达式节点);
            // 按文本前缀推断：数字/字符串/布尔
            if (字面量.值 == "真" || 字面量.值 == "假") return "布尔";
            if (!字面量.值.empty() && (字面量.值[0] == '"' || 字面量.值[0] == '\'')) return "字符串";
            return "整数";
        }
        case 节点类型::标识符表达式: {
            auto& 标识符 = static_cast<标识符表达式&>(表达式节点);
            auto 类型 = 符号表_.查找(标识符.名称);
            if (!类型) {
                诊断_.报告(诊断::错误(标识符.位置, "未定义符号 '" + 标识符.名称 + "'"));
                return "未知";
            }
            return *类型;
        }
        case 节点类型::二元运算: {
            return 检查二元运算(static_cast<二元运算&>(表达式节点));
        }
        default:
            return "未知";
    }
}

// 检查二元运算
std::string 语义分析器::检查二元运算(二元运算& 运算) {
    std::string 左类型 = 检查表达式(*运算.左);
    std::string 右类型 = 检查表达式(*运算.右);
    // 算术运算要求整型/浮点，比较与逻辑返回布尔
    switch (运算.运算) {
        case 运算符::加: case 运算符::减:
        case 运算符::乘: case 运算符::除: case 运算符::取余:
            if (!类型系统::可隐式转换(右类型, 左类型)) {
                诊断_.报告(诊断::错误(运算.位置,
                    "运算符操作数类型不匹配: '" + 左类型 + "' 与 '" + 右类型 + "'"));
            }
            return 左类型;
        case 运算符::等号等号: case 运算符::不等号:
        case 运算符::小于: case 运算符::大于:
        case 运算符::小于等于: case 运算符::大于等于:
        case 运算符::逻辑与: case 运算符::逻辑或:
            return "布尔";
        default:
            return 左类型;
    }
}

} // namespace cn_compiler

// IR生成器实现：AST降级为SSA形式IR（阶段一覆盖算术/调用/返回/常量）
#include "cn_compiler/ir/ir_generator.hpp"

namespace cn_compiler {

// 源码类型 -> IR类型映射（规格书7.5类型表示）
std::string IR生成器::类型映射(const std::string& 类型) {
    if (类型 == "整8") return "i8";
    if (类型 == "整16") return "i16";
    if (类型 == "整32" || 类型 == "整数") return "i32";
    if (类型 == "整64") return "i64";
    if (类型 == "整128") return "i128";
    if (类型 == "正8") return "u8";
    if (类型 == "正16") return "u16";
    if (类型 == "正32") return "u32";
    if (类型 == "正64") return "u64";
    if (类型 == "正128") return "u128";
    if (类型 == "浮32") return "f32";
    if (类型 == "浮64" || 类型 == "小数") return "f64";
    if (类型 == "布尔") return "i1";
    if (类型 == "字符" || 类型 == "字符串") return "ptr";
    if (类型 == "空类型") return "void";
    // 指针类型（类型* -> ptr）
    if (!类型.empty() && 类型.back() == '*') return "ptr";
    return "struct";  // 自定义类型（结构体/枚举/类）
}

// 新建虚拟寄存器
std::string IR生成器::新建虚拟寄存器() {
    return "%v" + std::to_string(虚拟寄存器计数_++);
}

// 添加指令
void IR生成器::添加指令(ir::基本块& 块, ir::操作码 操作码,
                         const std::vector<std::string>& 源操作数,
                         const std::string& 目标寄存器, const std::string& 类型,
                         const SourceLocation& 位置) {
    ir::指令 指令;
    指令.操作码 = 操作码;
    指令.源操作数 = 源操作数;
    指令.目标寄存器 = 目标寄存器;
    指令.类型 = 类型;
    指令.位置 = 位置;
    块.指令列表.push_back(指令);
}

// 生成函数
void IR生成器::生成函数(ir::IR模块& 模块, 函数声明& 函数) {
    ir::IR函数 ir函数;
    ir函数.名称 = 函数.名称;
    ir函数.返回类型 = 类型映射(函数.返回类型.empty() ? "空类型" : 函数.返回类型);
    // 参数列表
    int 参数编号 = 0;
    for (auto& 参数 : 函数.参数列表) {
        ir函数.参数列表.emplace_back(参数->名称, 类型映射(参数->类型));
        ir函数.参数编号映射[参数->名称] = 参数编号++;
    }
    // 入口基本块
    auto 入口块 = std::make_unique<ir::基本块>();
    入口块->名称 = "块0";
    // 函数体语句生成
    if (函数.函数体) {
        for (auto& 语句 : 函数.函数体->语句列表) {
            生成语句(模块, ir函数, *入口块, *语句);
        }
    }
    // 无返回语句时默认返回（空类型函数）
    if (ir函数.返回类型 == "void" && 入口块->终止类型.empty()) {
        入口块->终止类型 = "返回";
    }
    ir函数.基本块列表.push_back(std::move(入口块));
    模块.函数列表.push_back(std::move(ir函数));
}

// 生成语句（阶段一覆盖返回/表达式语句/变量声明）
void IR生成器::生成语句(ir::IR模块& 模块, ir::IR函数& 函数, ir::基本块& 当前块, 语句& 语句节点) {
    switch (语句节点.类型()) {
        case 节点类型::返回语句: {
            auto& 返回 = static_cast<返回语句&>(语句节点);
            if (返回.值) {
                当前块.返回值 = 生成表达式(模块, 函数, 当前块, *返回.值);
            }
            当前块.终止类型 = "返回";
            break;
        }
        case 节点类型::表达式语句: {
            auto& 表达式语句 = static_cast<cn_compiler::表达式语句&>(语句节点);
            生成表达式(模块, 函数, 当前块, *表达式语句.表达式);
            break;
        }
        case 节点类型::变量声明: {
            auto& 声明 = static_cast<变量声明&>(语句节点);
            if (声明.初始值) {
                // 变量声明生成常量定义指令（局部变量栈分配在阶段二完善）
                std::string 值寄存器 = 生成表达式(模块, 函数, 当前块, *声明.初始值);
                std::string 目标 = 新建虚拟寄存器();
                添加指令(当前块, ir::操作码::常量定义, {值寄存器}, 目标,
                         类型映射(声明.类型.empty() ? "整32" : 声明.类型), 声明.位置);
            }
            break;
        }
        default:
            // 控制流语句：阶段二实现
            break;
    }
}

// 生成表达式（阶段一覆盖字面量/标识符/二元运算/调用）
std::string IR生成器::生成表达式(ir::IR模块& 模块, ir::IR函数& 函数, ir::基本块& 当前块, 表达式& 表达式节点) {
    switch (表达式节点.类型()) {
        case 节点类型::字面量表达式: {
            auto& 字面量 = static_cast<字面量表达式&>(表达式节点);
            std::string 目标 = 新建虚拟寄存器();
            std::string 类型 = 类型映射("整32");  // 阶段一简化：数字按整32
            if (字面量.值 == "真" || 字面量.值 == "假") 类型 = "i1";
            添加指令(当前块, ir::操作码::常量定义, {字面量.值}, 目标, 类型, 字面量.位置);
            return 目标;
        }
        case 节点类型::标识符表达式: {
            auto& 标识符 = static_cast<标识符表达式&>(表达式节点);
            // 阶段一简化：参数直接引用虚拟寄存器（参数编号与虚拟寄存器编号对应）
            return "%v" + std::to_string(函数.参数编号映射[标识符.名称]);
        }
        case 节点类型::二元运算: {
            auto& 二元 = static_cast<二元运算&>(表达式节点);
            std::string 左 = 生成表达式(模块, 函数, 当前块, *二元.左);
            std::string 右 = 生成表达式(模块, 函数, 当前块, *二元.右);
            std::string 目标 = 新建虚拟寄存器();
            // 运算符 -> IR操作码映射（整型算术，浮点变体阶段二完善）
            ir::操作码 运算码;
            switch (二元.运算) {
                case 运算符::加: 运算码 = ir::操作码::加; break;
                case 运算符::减: 运算码 = ir::操作码::减; break;
                case 运算符::乘: 运算码 = ir::操作码::乘; break;
                case 运算符::除: 运算码 = ir::操作码::除; break;
                case 运算符::取余: 运算码 = ir::操作码::取余; break;
                default: 运算码 = ir::操作码::无操作; break;
            }
            添加指令(当前块, 运算码, {左, 右}, 目标, "i32", 二元.位置);
            return 目标;
        }
        case 节点类型::调用表达式: {
            auto& 调用 = static_cast<调用表达式&>(表达式节点);
            std::vector<std::string> 参数寄存器;
            for (auto& 参数 : 调用.参数列表) {
                参数寄存器.push_back(生成表达式(模块, 函数, 当前块, *参数));
            }
            std::string 目标 = 新建虚拟寄存器();
            添加指令(当前块, ir::操作码::直接调用, 参数寄存器, 目标, "i32", 调用.位置);
            // 记录被调用函数名（阶段一简化：指令源操作数首元素存函数名）
            当前块.指令列表.back().源操作数.insert(当前块.指令列表.back().源操作数.begin(), 调用.函数名);
            return 目标;
        }
        default:
            return "";
    }
}

// 主入口
ir::IR模块 IR生成器::生成(程序& 程序, 符号表& 符号表) {
    (void)符号表;  // 阶段一暂未使用符号表（参数编号映射简化）
    ir::IR模块 模块;
    虚拟寄存器计数_ = 0;
    for (auto& 声明 : 程序.顶层声明) {
        if (声明->类型() == 节点类型::函数声明) {
            生成函数(模块, static_cast<函数声明&>(*声明));
        }
    }
    return 模块;
}

} // namespace cn_compiler
```

> **实现说明：** `IR函数` 结构中已包含 `参数编号映射` 字段（见 `ir.hpp`），标识符表达式对参数的引用通过参数编号映射解析；局部变量的栈分配在阶段二实现。
