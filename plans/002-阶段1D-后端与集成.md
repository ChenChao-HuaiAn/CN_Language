创建 `src/cn_compiler/codegen/x64/x64_codegen.cpp`：

```cpp
// Win x64代码生成器实现：IR指令映射为x86-64汇编
#include "cn_compiler/codegen/x64/x64_codegen.hpp"
#include <sstream>

namespace cn_compiler {

// 参数编号 -> Win x64寄存器（前4个整型参数）
std::string X64代码生成器::参数寄存器(int 编号) const {
    switch (编号) {
        case 0: return "rcx";
        case 1: return "rdx";
        case 2: return "r8";
        case 3: return "r9";
        default: return "rsp+" + std::to_string(40 + (编号 - 4) * 8);  // 栈上参数
    }
}

// 名字修饰：中文符号 UTF-8十六进制编码（规格书附录C）
std::string X64代码生成器::名字修饰(const std::string& 名称) {
    // 简化实现：? + 十六进制编码（阶段一ASCII符号直接?名@@Y...@Z）
    std::stringstream 流;
    流 << "?";
    for (unsigned char c : 名称) {
        char 缓冲[8];
        snprintf(缓冲, sizeof(缓冲), "%02X", c);
        流 << 缓冲;
    }
    流 << "@@Y";
    return 流.str();
}

// 指令选择：IR操作码 -> x86-64助记符
std::string X64代码生成器::指令选择(ir::操作码 操作码, const std::string& 类型) const {
    // 32位整数运算使用 eax/ebx 等寄存器，浮点使用 xmm（阶段二完善）
    switch (操作码) {
        case ir::操作码::加: return "add";
        case ir::操作码::减: return "sub";
        case ir::操作码::乘: return "imul";
        case ir::操作码::除: return "idiv";
        case ir::操作码::取余: return "idiv";  // 余数在rdx，需额外处理
        default: return "; 未映射指令";
    }
}

// 生成单个函数汇编
std::string X64代码生成器::生成函数汇编(const ir::IR函数& 函数) {
    std::stringstream 汇编;
    // 函数声明（PROC/ENDP，规格书8.8示例）
    汇编 << "?函数@" << 名字修饰(函数.名称) << "@@Y" << " PROC\n";
    // prologue：保存rbp，建立栈帧
    汇编 << "    push    rbp\n";
    汇编 << "    mov     rbp, rsp\n";
    // 参数映射：前4个参数从寄存器读取到虚拟寄存器对应栈槽
    // （阶段一简化：参数直接使用寄存器，虚拟寄存器与参数编号一致）
    // 指令选择：遍历每个基本块
    for (auto& 块 : 函数.基本块列表) {
        汇编 << 块->名称 << ":\n";
        for (auto& 指令 : 块->指令列表) {
            std::string 助记符 = 指令选择(指令.操作码, 指令.类型);
            if (助记符[0] == ';') {
                汇编 << "    " << 助记符 << " ; " << 指令.目标寄存器 << " = "
                      << (指令.源操作数.empty() ? "" : 指令.源操作数[0]) << "\n";
                continue;
            }
            // 阶段一简化：目标/源操作数使用参数寄存器（rax/rcx/rdx）
            // 完整寄存器分配在阶段四实现
            if (!指令.源操作数.empty()) {
                汇编 << "    mov     eax, " << 参数寄存器(0) << "\n";
                汇编 << "    " << 助记符 << "     eax, " << 参数寄存器(1) << "\n";
            }
        }
        // 终止指令：返回
        if (块->终止类型 == "返回") {
            汇编 << "    mov     eax, eax\n";  // 阶段一：返回值在eax
        }
    }
    // epilogue
    汇编 << "    pop     rbp\n";
    汇编 << "    ret\n";
    汇编 << "ENDP\n";
    return 汇编.str();
}

// 主入口：生成完整汇编文件
std::string X64代码生成器::生成汇编(const ir::IR模块& 模块) {
    std::stringstream 汇编;
    汇编 << "; CN语言编译器生成的Win x64汇编\n";
    汇编 << "; 目标平台: win-x64\n";
    汇编 << ".code\n";
    for (const auto& 函数 : 模块.函数列表) {
        汇编 << 生成函数汇编(函数);
    }
    汇编 << "END\n";
    return 汇编.str();
}

} // namespace cn_compiler
```

> **实现说明：** 阶段一代码生成器采用最简策略（参数直通寄存器、返回值eax），验证IR→汇编链路贯通；完整寄存器分配、栈帧布局、溢出处理在阶段四（规格书8.3/8.4）实现。
