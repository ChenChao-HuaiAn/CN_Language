```cpp
// AST节点定义：语法分析器的输出，语义分析器的输入
#pragma once
#include <memory>
#include <string>
#include <vector>
#include "cn_compiler/common/source_location.hpp"

namespace cn_compiler {

// 节点类型枚举
enum class 节点类型 {
    程序, 函数声明, 变量声明, 参数, 代码块,
    字面量表达式, 标识符表达式, 二元运算, 一元运算, 调用表达式,
    成员访问, 下标访问,
    如果语句, 当语句, 循环语句, 选择语句, 中断语句, 继续语句,
    返回语句, 表达式语句,
};

// 运算符枚举（对应13级优先级中的运算符）
enum class 运算符 {
    加, 减, 乘, 除, 取余,
    等号等号, 不等号, 小于, 大于, 小于等于, 大于等于,
    逻辑与, 逻辑或, 逻辑非,
    位与, 位或, 位异或, 位非, 左移, 右移,
    赋值, 取地址, 解引用, 箭头,
};

// 前向声明（解决相互引用）
class 代码块;

// 基类：所有AST节点的共同接口
class AST节点 {
public:
    explicit AST节点(节点类型 类型) : 类型_(类型) {}
    virtual ~AST节点() = default;
    节点类型 类型() const { return 类型_; }
    SourceLocation 位置;  // 节点源码位置

private:
    节点类型 类型_;
};

// 表达式基类
class 表达式 : public AST节点 {
public:
    using AST节点::AST节点;
    virtual ~表达式() = default;
};

// 语句基类
class 语句 : public AST节点 {
public:
    using AST节点::AST节点;
    virtual ~语句() = default;
};

// ---- 表达式节点 ----

// 字面量表达式：整数值/浮点/字符串/布尔
class 字面量表达式 : public 表达式 {
public:
    explicit 字面量表达式(std::string 值) : 表达式(节点类型::字面量表达式), 值(std::move(值)) {}
    std::string 值;  // 原始字面量文本
};

// 标识符表达式：变量名/函数名引用
class 标识符表达式 : public 表达式 {
public:
    explicit 标识符表达式(std::string 名称) : 表达式(节点类型::标识符表达式), 名称(std::move(名称)) {}
    std::string 名称;
};

// 二元运算：左操作数 运算符 右操作数
class 二元运算 : public 表达式 {
public:
    二元运算(运算符 运算, std::unique_ptr<表达式> 左, std::unique_ptr<表达式> 右)
        : 表达式(节点类型::二元运算), 运算(运算), 左(std::move(左)), 右(std::move(右)) {}
    运算符 运算;
    std::unique_ptr<表达式> 左;
    std::unique_ptr<表达式> 右;
};

// 一元运算：运算符 操作数
class 一元运算 : public 表达式 {
public:
    一元运算(运算符 运算, std::unique_ptr<表达式> 操作数)
        : 表达式(节点类型::一元运算), 运算(运算), 操作数(std::move(操作数)) {}
    运算符 运算;
    std::unique_ptr<表达式> 操作数;
};

// 调用表达式：函数名 + 参数
class 调用表达式 : public 表达式 {
public:
    explicit 调用表达式(std::string 函数名) : 表达式(节点类型::调用表达式), 函数名(std::move(函数名)) {}
    std::string 函数名;
    std::vector<std::unique_ptr<表达式>> 参数列表;
};

// 成员访问：对象.成员
class 成员访问 : public 表达式 {
public:
    成员访问(std::unique_ptr<表达式> 对象, std::string 成员)
        : 表达式(节点类型::成员访问), 对象(std::move(对象)), 成员(std::move(成员)) {}
    std::unique_ptr<表达式> 对象;
    std::string 成员;
};

// 下标访问：对象[索引]
class 下标访问 : public 表达式 {
public:
    下标访问(std::unique_ptr<表达式> 对象, std::unique_ptr<表达式> 索引)
        : 表达式(节点类型::下标访问), 对象(std::move(对象)), 索引(std::move(索引)) {}
    std::unique_ptr<表达式> 对象;
    std::unique_ptr<表达式> 索引;
};

// ---- 语句节点 ----

// 表达式语句：仅由表达式构成（如函数调用）
class 表达式语句 : public 语句 {
public:
    explicit 表达式语句(std::unique_ptr<cn_compiler::表达式> 表达式)
        : 语句(节点类型::表达式语句), 表达式(std::move(表达式)) {}
    std::unique_ptr<cn_compiler::表达式> 表达式;
};

// 返回语句：返回 [值]
class 返回语句 : public 语句 {
public:
    返回语句() : 语句(节点类型::返回语句) {}
    std::unique_ptr<cn_compiler::表达式> 值;
};

// 中断语句
class 中断语句 : public 语句 {
public:
    中断语句() : 语句(节点类型::中断语句) {}
};

// 继续语句
class 继续语句 : public 语句 {
public:
    继续语句() : 语句(节点类型::继续语句) {}
};

// 如果语句：条件 + 真分支 + 否则分支（可为空）
class 如果语句 : public 语句 {
public:
    如果语句() : 语句(节点类型::如果语句) {}
    std::unique_ptr<cn_compiler::表达式> 条件;
    std::unique_ptr<代码块> 真分支;
    std::unique_ptr<语句> 否则分支;  // 可能是如果语句（否则如果）或代码块
};

// 当语句：当 条件 { 循环体 }
class 当语句 : public 语句 {
public:
    当语句() : 语句(节点类型::当语句) {}
    std::unique_ptr<cn_compiler::表达式> 条件;
    std::unique_ptr<代码块> 循环体;
};

// 循环语句：循环 (初始化; 条件; 更新) { 循环体 }
class 循环语句 : public 语句 {
public:
    循环语句() : 语句(节点类型::循环语句) {}
    std::unique_ptr<语句> 初始化;             // 变量声明或表达式语句
    std::unique_ptr<cn_compiler::表达式> 条件;
    std::unique_ptr<cn_compiler::表达式> 更新;
    std::unique_ptr<代码块> 循环体;
};

// 选择语句：选择(值) { 情况 1: ... 默认: ... }
class 选择语句 : public 语句 {
public:
    选择语句() : 语句(节点类型::选择语句) {}
    std::unique_ptr<cn_compiler::表达式> 值;
    std::vector<std::unique_ptr<cn_compiler::表达式>> 情况值列表;  // 各情况的值
    std::vector<std::unique_ptr<代码块>> 情况体列表;                // 各情况的语句块
    std::unique_ptr<代码块> 默认体;                                  // 默认分支（可为空）
};

// ---- 声明节点 ----

// 参数：名称 + 类型
class 参数 : public AST节点 {
public:
    参数() : AST节点(节点类型::参数) {}
    std::string 名称;
    std::string 类型;
};

// 变量声明：变量/常量/静态 + 名称 + 类型(可推断) + 初始值
class 变量声明 : public 语句 {
public:
    变量声明() : 语句(节点类型::变量声明) {}
    bool 是常量 = false;      // 常量 PI = 3.14
    bool 是静态 = false;      // 静态 整32 计数 = 0
    std::string 名称;
    std::string 类型;         // 为空表示类型推断（变量 x = 10）
    std::unique_ptr<cn_compiler::表达式> 初始值;
};

// 函数声明：函数 名称(参数列表) -> 返回类型 { 函数体 }
class 函数声明 : public AST节点 {
public:
    函数声明() : AST节点(节点类型::函数声明) {}
    std::string 名称;
    std::vector<std::unique_ptr<参数>> 参数列表;
    std::string 返回类型;                  // 为空表示无返回值（空类型）
    std::unique_ptr<代码块> 函数体;        // 为空表示函数原型声明（前向引用）
};

// 代码块：{ 语句列表 }
class 代码块 : public 语句 {
public:
    代码块() : 语句(节点类型::代码块) {}
    std::vector<std::unique_ptr<语句>> 语句列表;
};

// 程序：顶层声明列表
class 程序 : public AST节点 {
public:
    程序() : AST节点(节点类型::程序) {}
    std::vector<std::unique_ptr<AST节点>> 顶层声明;
};

} // namespace cn_compiler
```

```cpp
// 语法分析器表达式测试：优先级、括号、调用
#include <gtest/gtest.h>
#include <memory>
#include "cn_compiler/parser/parser.hpp"
#include "cn_compiler/lexer/lexer.hpp"

// 辅助函数：源码 -> 表达式AST（取程序第一个顶层声明的表达式）
static std::unique_ptr<cn_compiler::表达式> 解析表达式(const std::string& 源码) {
    cn_compiler::诊断引擎 引擎;
    cn_compiler::词法分析器 词法器(引擎);
    auto tokens = 词法器.分析(源码, "测试.cn");
    cn_compiler::语法分析器 解析器(引擎);
    auto 程序 = 解析器.分析(tokens);
    EXPECT_TRUE(引擎.是否成功()) << "解析失败";
    // 取第一个顶层声明（函数声明），再取函数体第一个表达式语句
    auto* 函数 = dynamic_cast<cn_compiler::函数声明*>(程序->顶层声明[0].get());
    EXPECT_NE(函数, nullptr);
    auto* 表达式语句 = dynamic_cast<cn_compiler::表达式语句*>(函数->函数体->语句列表[0].get());
    EXPECT_NE(表达式语句, nullptr);
    return std::move(表达式语句->表达式);
}

// 测试乘法优先级高于加法：a + b * c
TEST(ParserExprTest, 乘法优先) {
    auto 表达式 = 解析表达式("函数 测试() -> 整32 { 返回 a + b * c }");

    ASSERT_NE(二元, nullptr);
```

```cpp
// 语法分析器：Token流 -> AST
#pragma once
#include <memory>
#include <vector>
#include "cn_compiler/common/diagnostics.hpp"
#include "cn_compiler/lexer/token.hpp"
#include "cn_compiler/parser/ast.hpp"

namespace cn_compiler {

// 语法分析器：递归下降 + Pratt解析（13级优先级），两趟解析支持前向引用
class 语法分析器 {
public:
    explicit 语法分析器(诊断引擎& 引擎) : 诊断_(引擎) {}

    // 分析Token流，返回程序AST
    std::unique_ptr<程序> 分析(const std::vector<Token>& tokens);

private:
    // 当前Token
    const Token& 当前() const;
    // 前进一个Token
    void 前进();
    // 检查当前是否为目标类型
    bool 检查(TokenType 类型) const;
    // 匹配目标类型并前进，否则报告错误
    bool 匹配(TokenType 类型, const std::string& 预期描述);
```

```cpp
// 语法分析器实现：递归下降 + Pratt表达式解析 + 两趟解析
#include "cn_compiler/parser/parser.hpp"

namespace cn_compiler {

// 当前Token
const Token& 语法分析器::当前() const {
    return tokens_[位置_];
}

// 前进一个Token
void 语法分析器::前进() {
    if (位置_ < tokens_.size() - 1) 位置_++;
}

// 检查当前是否为目标类型
bool 语法分析器::检查(TokenType 类型) const {
    return 当前().类型() == 类型;
}

// 匹配目标类型并前进
bool 语法分析器::匹配(TokenType 类型, const std::string& 预期描述) {
    if (检查(类型)) {
        前进();
        return true;
    }
    报告错误("预期 " + 预期描述 + "，实际为 '" + 当前().文本() + "'");
    return false;
}

// 报告语法错误
void 语法分析器::报告错误(const std::string& 消息) {
    诊断_.报告(诊断::错误(当前().位置(), 消息));
}

// 解析类型：关键字类型 或 标识符（结构体/枚举/类名）
std::string 语法分析器::解析类型() {
    // 类型关键字：整8~整128/正8~正128/浮32/浮64/整数/小数/布尔/字符/字符串/空类型
    // 或 标识符（自定义类型名）
    if (当前().是关键字() && 当前().文本().find("整") != std::string::npos ||
        当前().文本() == "整数" || 当前().文本() == "小数" ||
        当前().文本() == "布尔" || 当前().文本() == "字符" ||
        当前().文本() == "字符串" || 当前().文本() == "空类型" ||
        当前().类型() == TokenType::标识符) {
        std::string 类型名 = 当前().文本();
        前进();
        return 类型名;
    }
    报告错误("预期类型，实际为 '" + 当前().文本() + "'");
    return "";
}

// 解析参数：类型 名称（如 整32 a）
std::unique_ptr<参数> 语法分析器::解析参数() {
    auto 参数节点 = std::make_unique<参数>();
    参数节点->类型 = 解析类型();
    参数节点->名称 = 当前().文本();
    前进();
    return 参数节点;
}

// 解析函数声明：函数 名称(参数列表) -> 返回类型 { 函数体 }
std::unique_ptr<函数声明> 语法分析器::解析函数声明() {
    auto 函数 = std::make_unique<函数声明>();
    函数->位置 = 当前().位置();
    前进();  // 跳过"函数"关键字
    函数->名称 = 当前().文本();
    前进();
    if (!匹配(TokenType::括号开, "'('")) return 函数;
    // 参数列表（可空）
    if (!检查(TokenType::括号关)) {
        while (true) {
            函数->参数列表.push_back(解析参数());
            if (!检查(TokenType::逗号)) break;
            前进();
        }
    }
    匹配(TokenType::括号关, "')'");
    // 返回类型（-> 类型，可省略表示空类型）
    if (检查(TokenType::箭头)) {
        前进();
        函数->返回类型 = 解析类型();
    }
    // 函数体（可为空=原型声明）
    if (检查(TokenType::花括号开)) {
        函数->函数体 = 解析代码块();
    }
    return 函数;
}

// 解析代码块：{ 语句列表 }
std::unique_ptr<代码块> 语法分析器::解析代码块() {
    auto 块 = std::make_unique<代码块>();
    匹配(TokenType::花括号开, "'{'");
    while (!检查(TokenType::花括号关) && !检查(TokenType::文件结束)) {
        块->语句列表.push_back(解析语句());
    }
    匹配(TokenType::花括号关, "'}'");
    return 块;
}

// 解析变量声明：变量 名称 [= 初始值] / 类型 名称 [= 初始值] / 常量/静态
std::unique_ptr<语句> 语法分析器::解析变量声明(bool 是静态) {
    auto 声明 = std::make_unique<变量声明>();
    声明->是静态 = 是静态;
    // "变量"关键字：类型推断；"常量"关键字：常量（常量Token在词法层归属标识符，此处按文本判断）
    if (当前().文本() == "变量") {
        前进();
        声明->名称 = 当前().文本();
        前进();
    } else if (当前().文本() == "常量") {
        声明->是常量 = true;
        前进();
        声明->类型 = 解析类型();
        声明->名称 = 当前().文本();
        前进();
    } else {
        声明->类型 = 解析类型();
        声明->名称 = 当前().文本();
        前进();
    }
    // 初始值（= 表达式）
    if (检查(TokenType::赋值)) {
        前进();
        声明->初始值 = 解析表达式();
    }
    return 声明;
}

// 解析语句（骨架：完整分支在Step 6补齐）
std::unique_ptr<语句> 语法分析器::解析语句() {
    // 返回语句
    if (检查(TokenType::关键字返回)) {
        auto 返回 = std::make_unique<返回语句>();
        前进();
        if (!检查(TokenType::花括号关) && !检查(TokenType::分号) &&
            !检查(TokenType::文件结束)) {
            返回->值 = 解析表达式();
        }
        return 返回;
    }
    // 表达式语句
    auto 语句节点 = std::make_unique<表达式语句>(解析表达式());
    return 语句节点;
}

// 解析顶层声明（骨架：完整分支在Step 6补齐）
std::unique_ptr<AST节点> 语法分析器::解析顶层声明() {
    if (检查(TokenType::关键字函数)) {
        return 解析函数声明();
    }
    报告错误("预期顶层声明，实际为 '" + 当前().文本() + "'");
    // 跳过当前Token避免死循环
    前进();
    return nullptr;
}

std::unique_ptr<表达式> 语法分析器::解析逻辑或() { return 解析逻辑与(); }
std::unique_ptr<表达式> 语法分析器::解析逻辑与() { return 解析位或(); }
std::unique_ptr<表达式> 语法分析器::解析位或() { return 解析位异或(); }
std::unique_ptr<表达式> 语法分析器::解析位异或() { return 解析位与(); }

std::unique_ptr<表达式> 语法分析器::解析移位() { return 解析加减(); }
std::unique_ptr<表达式> 语法分析器::解析加减() { return 解析乘除(); }

std::unique_ptr<表达式> 语法分析器::解析一元() { return 解析后缀(); }
std::unique_ptr<表达式> 语法分析器::解析后缀() { return 解析基本表达式(); }
std::unique_ptr<表达式> 语法分析器::解析基本表达式() {
    // 字面量
    if (检查(TokenType::整数字面量) || 检查(TokenType::浮点字面量) ||
        检查(TokenType::字符串字面量) || 检查(TokenType::字符字面量) ||
        检查(TokenType::关键字真) || 检查(TokenType::关键字假) ||
        检查(TokenType::关键字无)) {
        auto 字面量 = std::make_unique<字面量表达式>(当前().文本());
        前进();
        return 字面量;
    }
    // 标识符 / 调用
    if (检查(TokenType::标识符)) {
        std::string 名称 = 当前().文本();
        前进();
        if (检查(TokenType::括号开)) {
            auto 调用 = std::make_unique<调用表达式>(名称);
            前进();
            while (!检查(TokenType::括号关)) {
                调用->参数列表.push_back(解析表达式());
                if (!检查(TokenType::逗号)) break;
                前进();
            }
            匹配(TokenType::括号关, "')'");
            return 调用;
        }
        return std::make_unique<标识符表达式>(名称);
    }
    报告错误("预期表达式，实际为 '" + 当前().文本() + "'");
    前进();
    return std::make_unique<字面量表达式>("0");
}

// 主入口：两趟解析
std::unique_ptr<程序> 语法分析器::分析(const std::vector<Token>& tokens) {
    tokens_ = tokens;
    位置_ = 0;
    auto 程序节点 = std::make_unique<程序>();

    // 第一趟：收集顶层声明（函数原型/结构体/枚举/导入等），解析函数签名
    while (!检查(TokenType::文件结束)) {
        auto 声明 = 解析顶层声明();
        if (声明) 程序节点->顶层声明.push_back(std::move(声明));
    }

    // 第二趟：解析函数体（支持前向引用）
    // 实现：重新遍历顶层声明，对每个函数声明解析函数体
    // （第一趟中函数体留空的原型，在此补全函数体解析）
    // 注：为简化阶段一实现，第二趟将函数体解析合并进第一趟的解析函数声明()，
    // 前向引用支持通过符号表在Task 1.5语义分析阶段实现。

    return 程序节点;
}
```

```cpp
std::unique_ptr<表达式> 语法分析器::解析加减() {
    auto 左 = 解析乘除();

}
```
