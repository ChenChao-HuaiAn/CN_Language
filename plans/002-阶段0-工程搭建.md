```cpp
// 源码位置：记录文件名、行号、列号，供诊断系统使用
#pragma once
#include <string>

namespace cn_compiler {

// 源码位置类：所有Token和AST节点都携带位置信息用于错误报告
class SourceLocation {
public:
    SourceLocation() = default;
    SourceLocation(std::string 文件, int 行, int 列)
        : 文件_(std::move(文件)), 行_(行), 列_(列) {}

    // 获取文件名
    const std::string& 文件() const { return 文件_; }
    // 获取行号
    int 行() const { return 行_; }
    // 获取列号
    int 列() const { return 列_; }
    // 格式化为 "文件:行:列"
    std::string 字符串() const {
        if (文件_.empty()) return "未知位置";
        return 文件_ + ":" + std::to_string(行_) + ":" + std::to_string(列_);
    }

private:
    std::string 文件_;  // 源文件名
    int 行_ = 0;        // 行号（从1开始）
    int 列_ = 0;        // 列号（从1开始）
};

} // namespace cn_compiler
```

```cpp
// 诊断系统：统一的错误/警告/信息报告，含源码位置
#pragma once
#include <string>
#include <vector>
#include "cn_compiler/common/source_location.hpp"

namespace cn_compiler {

// 诊断级别
enum class 诊断级别 { 错误, 警告, 信息 };

// 单条诊断信息
class 诊断 {
public:
    诊断(诊断级别 级别, SourceLocation 位置, std::string 消息)
        : 级别_(级别), 位置_(位置), 消息_(std::move(消息)) {}

    // 便捷构造：错误
    static 诊断 错误(SourceLocation 位置, std::string 消息) {
        return 诊断(诊断级别::错误, 位置, std::move(消息));
    }
    // 便捷构造：错误（按行列）
    static 诊断 错误(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::错误, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }
    // 便捷构造：警告
    static 诊断 警告(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::警告, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }
    // 便捷构造：信息
    static 诊断 信息(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::信息, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }

    // 获取级别
    诊断级别 级别() const { return 级别_; }
    // 获取位置
    const SourceLocation& 位置() const { return 位置_; }
    // 获取消息
    const std::string& 消息() const { return 消息_; }

private:
    诊断级别 级别_;          // 级别
    SourceLocation 位置_;    // 源码位置
    std::string 消息_;       // 诊断消息
};

// 诊断引擎：收集所有诊断并统计错误/警告数量
class 诊断引擎 {
public:
    // 报告一条诊断
    void 报告(const 诊断& 诊断) {
        诊断列表_.push_back(诊断);
        if (诊断.级别() == 诊断级别::错误) 错误计数_++;
        if (诊断.级别() == 诊断级别::警告) 警告计数_++;
    }
    // 是否成功（无错误）
    bool 是否成功() const { return 错误计数_ == 0; }
    // 错误数量
    int 错误计数() const { return 错误计数_; }
    // 警告数量
    int 警告计数() const { return 警告计数_; }
    // 获取全部诊断
    const std::vector<诊断>& 全部() const { return 诊断列表_; }

private:
    std::vector<诊断> 诊断列表_;  // 诊断列表
    int 错误计数_ = 0;            // 错误数量
    int 警告计数_ = 0;            // 警告数量
};

} // namespace cn_compiler
```

```cpp
// CN语言编译器命令行入口
#include <iostream>
#include <string>
#include <vector>
#include "cn_compiler/common/diagnostics.hpp"

// 打印版本信息
void 打印版本() {
    std::cout << "CN语言编译器 cn 0.1.0\n";
    std::cout << "目标平台: win-x64 (初期), linux-arm64 (初期)\n";
}

// 打印帮助
void 打印帮助() {
    std::cout << "用法: cn <命令> [选项] 源文件...\n";
    std::cout << "命令:\n";
    std::cout << "  build    编译并链接生成可执行文件\n";
    std::cout << "  compile  仅编译生成汇编\n";
    std::cout << "  run      编译并运行\n";
    std::cout << "  check    仅做语法与语义检查\n";
    std::cout << "  ir       生成并输出IR\n";
    std::cout << "  ast      输出AST\n";
    std::cout << "  token    输出Token流\n";
    std::cout << "选项:\n";
    std::cout << "  --target <目标>   目标平台 (win-x64 / linux-arm64)\n";
    std::cout << "  --opt <级别>      优化级别 (0/1/2/3, 默认2)\n";
    std::cout << "  --output <文件>   输出文件路径\n";
    std::cout << "  --版本 / --帮助   显示版本/帮助\n";
}

// 未实现的命令处理（阶段零仅支持版本/帮助，后续阶段逐个填充）
int 未实现命令(const std::string& 命令) {
    std::cerr << "错误: 命令 '" << 命令 << "' 尚未实现（请参考实施计划对应阶段）\n";
    return 1;
}

// 程序入口
int main(int 参数个数, char** 参数值) {
    std::vector<std::string> 参数(参数值 + 1, 参数值 + 参数个数);

    // 无参数：打印帮助
    if (参数.empty()) {
        打印帮助();
        return 0;
    }

    // 解析第一个参数为命令
    const std::string 命令 = 参数[0];
    if (命令 == "--版本" || 命令 == "-v") {
        打印版本();
        return 0;
    }
    if (命令 == "--帮助" || 命令 == "-h") {
        打印帮助();
        return 0;
    }
    if (命令 == "build" || 命令 == "compile" || 命令 == "run" ||
        命令 == "check" || 命令 == "ir" || 命令 == "ast" || 命令 == "token") {
        return 未实现命令(命令);
    }

    // 未知命令
    std::cerr << "错误: 未知命令 '" << 命令 << "'\n";
    打印帮助();
    return 1;
}
```

```python
#!/usr/bin/env python3
# CN语言E2E测试运行器：编译->链接->运行->比对输出
# 用法: python3 run_e2e.py --cn <编译器路径>
import argparse
import os
import pathlib
import subprocess
import sys

# E2E用例根目录
用例根目录 = pathlib.Path(__file__).parent


def 运行命令(命令列表: list, 工作目录: pathlib.Path) -> subprocess.CompletedProcess:
    """执行命令并返回结果"""
    return subprocess.run(命令列表, cwd=str(工作目录), capture_output=True, text=True)


def 执行单个用例(编译器路径: pathlib.Path, 用例目录: pathlib.Path) -> tuple:
    """执行单个E2E用例：编译->运行->比对。返回(是否通过, 失败原因)"""
    cn文件 = list(用例目录.glob("*.cn"))[0]      # 用例源码
    期望文件 = 用例目录 / "hello.expected"         # 期望输出
    输出可执行 = pathlib.Path("target") / f"{用例目录.name}.exe"

    # 1. 编译
    编译结果 = 运行命令([str(编译器路径), "build", str(cn文件),
                      "--output", str(输出可执行)], 用例根目录)
    if 编译结果.returncode != 0:
        return False, f"编译失败: {编译结果.stderr}"

    # 2. 运行
    运行结果 = 运行命令([str(输出可执行)], 用例根目录)
    if 运行结果.returncode != 0:
        return False, f"运行失败(退出码{运行结果.returncode}): {运行结果.stderr}"

    # 3. 比对期望输出（逐行比对，忽略末尾空白差异）
    期望 = 期望文件.read_text(encoding="utf-8").splitlines()
    实际 = 运行结果.stdout.splitlines()
    if 实际 != 期望:
        return False, f"输出不一致\n期望: {期望}\n实际: {实际}"
    return True, ""


def 主程序() -> int:
    解析器 = argparse.ArgumentParser(description="CN语言E2E测试运行器")
    解析器.add_argument("--cn", required=True, help="cn编译器可执行文件路径")
    参数 = 解析器.parse_args()
    编译器路径 = pathlib.Path(参数.cn).resolve()

    # 收集所有E2E用例目录（形如 NN_名称）
    用例目录们 = sorted(
        [d for d in 用例根目录.iterdir() if d.is_dir() and d.name[:2].isdigit()],
        key=lambda d: d.name,
    )
    if not 用例目录们:
        print("未找到E2E用例目录")
        return 1

    通过数 = 0
    失败列表 = []
    for 用例目录 in 用例目录们:
        print(f"运行用例: {用例目录.name}")
        通过, 原因 = 执行单个用例(编译器路径, 用例目录)
        if 通过:
            通过数 += 1
            print(f"  ✓ 通过")
        else:
            失败列表.append((用例目录.name, 原因))
            print(f"  ✗ 失败: {原因}")

    print(f"\nE2E结果: {通过数}/{len(用例目录们)} 通过")
    if 失败列表:
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(主程序())
```
