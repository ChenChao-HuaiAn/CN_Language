# CN语言编译器实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: 使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现本计划。步骤使用复选框（`- [ ]`）语法跟踪进度。

> 基于设计规格书：[`plans/001 CN语言编译器设计规格书.md`](plans/001%20CN语言编译器设计规格书.md)

**目标：** 用C++从零编写CN语言编译器，直接生成汇编代码，实现自举

**架构：** 6阶段编译流水线（词法→语法→语义→IR→优化→代码生成），多趟+自定义IR；前端（frontend）→ IR → 优化器（opt）→ 代码生成（codegen）单向依赖，driver是唯一组合入口

**技术栈：** C++17、CMake、Google Test、NASM（Windows汇编器）、Python 3（E2E测试脚本）

**项目结构约定：** 本计划采用 [`plans/001`](plans/001%20CN语言编译器设计规格书.md) 第十三章定义的项目结构（`src/cn_compiler/...`），源文件路径与规格书保持一致，模块依赖约束：`frontend → ir → opt → codegen`（禁止反向），`driver` 依赖所有模块

---

## 全局约束

- 每个源文件不超过1000行，每个函数不超过100行（超出则拆分子模块目录，旧文件删除）
- 代码添加中文注释
- E2E先行：每个功能必须先写E2E测试，测试通过后才算完成
- 禁止skip()，所有测试必须真正运行，使用真实配置不使用Mock
- 编译产物放在 target/ 目录
- CMake构建系统，C++17标准，编译零警告
- 目标平台：初期 Win x64（NASM/ml64）+ Linux ARM64（GAS/as）
- 全部53个关键字均为保留字，不可用作标识符
- 内存管理：指针+手动内存管理（malloc/free式），运行时API（分配/释放/重新分配/复制内存/置零内存）
- 错误处理：错误码+可选值（结果<T,E>/可选<T>），无异常关键字，强制检查规则3条
- 类型命名：中文位宽命名（整8~整128/正8~正128/浮32/浮64）
- 字面量：支持Unicode字符转义 `'\u{4E2D}'`、原始字符串 `原始"..."`、多行字符串 `多行"""..."""`，不支持数字下划线
- 入口文件约定：`主.cn` 定义 `主` 函数，签名 `函数 主() -> 整32`（返回0表示成功）
- 每个任务结束：单元测试通过 + E2E测试通过 + CMake零警告 + 更新plans文档打勾 + Git提交（中文提交信息）

```cpp
// 源码位置：记录文件名、行号、列号，供诊断系统使用
#pragma once
#include <string>

namespace cn_compiler {

// 源码位置类：所有Token和AST节点都携带位置信息用于错误报告
class SourceLocation {
public:
    SourceLocation() = default;
    SourceLocation(std::string 文件, int 行, int 列)
        : 文件_(std::move(文件)), 行_(行), 列_(列) {}

    // 获取文件名
    const std::string& 文件() const { return 文件_; }
    // 获取行号
    int 行() const { return 行_; }
    // 获取列号
    int 列() const { return 列_; }
    // 格式化为 "文件:行:列"
    std::string 字符串() const {
        if (文件_.empty()) return "未知位置";
        return 文件_ + ":" + std::to_string(行_) + ":" + std::to_string(列_);
    }

private:
    std::string 文件_;  // 源文件名
    int 行_ = 0;        // 行号（从1开始）
    int 列_ = 0;        // 列号（从1开始）
};

} // namespace cn_compiler
```

```cpp
// 诊断系统：统一的错误/警告/信息报告，含源码位置
#pragma once
#include <string>
#include <vector>
#include "cn_compiler/common/source_location.hpp"

namespace cn_compiler {

// 诊断级别
enum class 诊断级别 { 错误, 警告, 信息 };

// 单条诊断信息
class 诊断 {
public:
    诊断(诊断级别 级别, SourceLocation 位置, std::string 消息)
        : 级别_(级别), 位置_(位置), 消息_(std::move(消息)) {}

    // 便捷构造：错误
    static 诊断 错误(SourceLocation 位置, std::string 消息) {
        return 诊断(诊断级别::错误, 位置, std::move(消息));
    }
    // 便捷构造：错误（按行列）
    static 诊断 错误(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::错误, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }
    // 便捷构造：警告
    static 诊断 警告(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::警告, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }
    // 便捷构造：信息
    static 诊断 信息(std::string 文件, int 行, int 列, std::string 消息) {
        return 诊断(诊断级别::信息, SourceLocation(std::move(文件), 行, 列), std::move(消息));
    }

    // 获取级别
    诊断级别 级别() const { return 级别_; }
    // 获取位置
    const SourceLocation& 位置() const { return 位置_; }
    // 获取消息
    const std::string& 消息() const { return 消息_; }

private:
    诊断级别 级别_;          // 级别
    SourceLocation 位置_;    // 源码位置
    std::string 消息_;       // 诊断消息
};

// 诊断引擎：收集所有诊断并统计错误/警告数量
class 诊断引擎 {
public:
    // 报告一条诊断
    void 报告(const 诊断& 诊断) {
        诊断列表_.push_back(诊断);
        if (诊断.级别() == 诊断级别::错误) 错误计数_++;
        if (诊断.级别() == 诊断级别::警告) 警告计数_++;
    }
    // 是否成功（无错误）
    bool 是否成功() const { return 错误计数_ == 0; }
    // 错误数量
    int 错误计数() const { return 错误计数_; }
    // 警告数量
    int 警告计数() const { return 警告计数_; }
    // 获取全部诊断
    const std::vector<诊断>& 全部() const { return 诊断列表_; }

private:
    std::vector<诊断> 诊断列表_;  // 诊断列表
    int 错误计数_ = 0;            // 错误数量
    int 警告计数_ = 0;            // 警告数量
};

} // namespace cn_compiler
```

创建 `src/cn_main.cpp`（CLI入口，解析 build/compile/run/check/ir/ast/token 子命令与 --target/--opt/--output 选项）：

```cpp
// CN语言编译器命令行入口
#include <iostream>
#include <string>
#include <vector>
#include "cn_compiler/common/diagnostics.hpp"

// 打印版本信息
void 打印版本() {
    std::cout << "CN语言编译器 cn 0.1.0\n";
    std::cout << "目标平台: win-x64 (初期), linux-arm64 (初期)\n";
}

// 打印帮助
void 打印帮助() {
    std::cout << "用法: cn <命令> [选项] 源文件...\n";
    std::cout << "命令:\n";
    std::cout << "  build    编译并链接生成可执行文件\n";
    std::cout << "  compile  仅编译生成汇编\n";
    std::cout << "  run      编译并运行\n";
    std::cout << "  check    仅做语法与语义检查\n";
    std::cout << "  ir       生成并输出IR\n";
    std::cout << "  ast      输出AST\n";
    std::cout << "  token    输出Token流\n";
    std::cout << "选项:\n";
    std::cout << "  --target <目标>   目标平台 (win-x64 / linux-arm64)\n";
    std::cout << "  --opt <级别>      优化级别 (0/1/2/3, 默认2)\n";
    std::cout << "  --output <文件>   输出文件路径\n";
    std::cout << "  --版本 / --帮助   显示版本/帮助\n";
}

// 未实现的命令处理（阶段零仅支持版本/帮助，后续阶段逐个填充）
int 未实现命令(const std::string& 命令) {
    std::cerr << "错误: 命令 '" << 命令 << "' 尚未实现（请参考实施计划对应阶段）\n";
    return 1;
}

// 程序入口
int main(int 参数个数, char** 参数值) {
    std::vector<std::string> 参数(参数值 + 1, 参数值 + 参数个数);

    // 无参数：打印帮助
    if (参数.empty()) {
        打印帮助();
        return 0;
    }

    // 解析第一个参数为命令
    const std::string 命令 = 参数[0];
    if (命令 == "--版本" || 命令 == "-v") {
        打印版本();
        return 0;
    }
    if (命令 == "--帮助" || 命令 == "-h") {
        打印帮助();
        return 0;
    }
    if (命令 == "build" || 命令 == "compile" || 命令 == "run" ||
        命令 == "check" || 命令 == "ir" || 命令 == "ast" || 命令 == "token") {
        return 未实现命令(命令);
    }

    // 未知命令
    std::cerr << "错误: 未知命令 '" << 命令 << "'\n";
    打印帮助();
    return 1;
}
```

```python
# E2E用例根目录
用例根目录 = pathlib.Path(__file__).parent


def 运行命令(命令列表: list, 工作目录: pathlib.Path) -> subprocess.CompletedProcess:
    """执行命令并返回结果"""
    return subprocess.run(命令列表, cwd=str(工作目录), capture_output=True, text=True)


def 执行单个用例(编译器路径: pathlib.Path, 用例目录: pathlib.Path) -> tuple:
    """执行单个E2E用例：编译->运行->比对。返回(是否通过, 失败原因)"""
    cn文件 = list(用例目录.glob("*.cn"))[0]      # 用例源码
    期望文件 = 用例目录 / "hello.expected"         # 期望输出
    输出可执行 = pathlib.Path("target") / f"{用例目录.name}.exe"

    # 1. 编译
    编译结果 = 运行命令([str(编译器路径), "build", str(cn文件),
                      "--output", str(输出可执行)], 用例根目录)
    if 编译结果.returncode != 0:
        return False, f"编译失败: {编译结果.stderr}"

    # 2. 运行
    运行结果 = 运行命令([str(输出可执行)], 用例根目录)
    if 运行结果.returncode != 0:
        return False, f"运行失败(退出码{运行结果.returncode}): {运行结果.stderr}"

    # 3. 比对期望输出（逐行比对，忽略末尾空白差异）
    期望 = 期望文件.read_text(encoding="utf-8").splitlines()
    实际 = 运行结果.stdout.splitlines()
    if 实际 != 期望:
        return False, f"输出不一致\n期望: {期望}\n实际: {实际}"
    return True, ""


def 主程序() -> int:
    解析器 = argparse.ArgumentParser(description="CN语言E2E测试运行器")
    解析器.add_argument("--cn", required=True, help="cn编译器可执行文件路径")
    参数 = 解析器.parse_args()
    编译器路径 = pathlib.Path(参数.cn).resolve()

    # 收集所有E2E用例目录（形如 NN_名称）
    用例目录们 = sorted(
        [d for d in 用例根目录.iterdir() if d.is_dir() and d.name[:2].isdigit()],
        key=lambda d: d.name,
    )
    if not 用例目录们:
        print("未找到E2E用例目录")
        return 1

    通过数 = 0
    失败列表 = []
    for 用例目录 in 用例目录们:

        else:
            失败列表.append((用例目录.name, 原因))
            print(f"  ✗ 失败: {原因}")

    print(f"\nE2E结果: {通过数}/{len(用例目录们)} 通过")
    if 失败列表:
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(主程序())
```

```cpp
    // ---- 错误处理关键字(2) ----
    关键字结果, 关键字可选,
    // ---- 字面量前缀关键字(2) ----

public:

        : 类型_(类型), 文本_(std::move(文本)), 位置_(位置) {}

    // 获取类型
    TokenType 类型() const { return 类型_; }
    // 获取文本

    // 获取源码位置
    const SourceLocation& 位置() const { return 位置_; }

    // 设置整数值

    // 判断是否是关键字之一
    bool 是关键字() const {
        return 类型_ >= TokenType::关键字如果 && 类型_ <= TokenType::关键字多行;
    }

private:
    TokenType 类型_;          // Token类型
    std::string 文本_;        // Token原始文本
    SourceLocation 位置_;     // 源码位置
    std::int64_t 整数值_ = 0; // 整数字面量值
};

} // namespace cn_compiler
```

```cpp
        {"无", TokenType::关键字无},
        // OOP
        {"类", TokenType::关键字类}, {"接口", TokenType::关键字接口},

        // 字面量前缀
        {"原始", TokenType::关键字原始}, {"多行", TokenType::关键字多行},
    };
    return 表;
}

// 查找关键字：命中返回true并设置类型
bool 词法分析器::查找关键字(const std::string& 文本, TokenType& 类型) {
    const auto& 表 = 关键字表();
    auto 迭代 = 表.find(文本);
    if (迭代 != 表.end()) {
        类型 = 迭代->second;
        return true;
    }
    return false;
}

// UTF-8解码：从位置_读取一个字符，返回Unicode码点
char32_t 词法分析器::前进() {
    if (到达末尾()) return 0;
    if (回退标记_) {
        回退标记_ = false;
        列_++;
        return 回退字符_;
    }
    unsigned char 首字节 = static_cast<unsigned char>(源码_[位置_]);
    位置_++;
    // 计算UTF-8字符长度并解码
    int 长度 = 1;
    char32_t 码点 = 首字节;
    if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
    else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
    else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
    for (int i = 1; i < 长度 && 位置_ < 源码_.size(); i++) {
        码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[位置_]) & 0x3F);
        位置_++;
    }
    列_++;
    return 码点;
}

// 回退一个字符
void 词法分析器::回退() {
    回退标记_ = true;
    // 将回退字符存于回退字符_，此处通过查看()预先取得
    // 简化实现：回退时列号递减由查看/前进共同维护
}

// 查看当前字符
char32_t 词法分析器::查看() const {
    if (到达末尾()) return 0;
    if (回退标记_) return 回退字符_;
    // 解码当前位置的字符
    unsigned char 首字节 = static_cast<unsigned char>(源码_[位置_]);
    if (首字节 < 0x80) return 首字节;
    int 长度 = 1;
    char32_t 码点 = 首字节;
    if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
    else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
    else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
    for (int i = 1; i < 长度 && 位置_ + i < 源码_.size(); i++) {
        码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[位置_ + i]) & 0x3F);
    }
    return 码点;
}

// 查看向后第n个字符
char32_t 词法分析器::查看第N个(int n) const {
    if (n <= 0) return 查看();
    if (回退标记_ && n == 1) {
        // 回退场景简化：仅支持n=1时返回回退字符
        return 回退字符_;
    }
    // 简化实现：从当前位置顺序解码n个字符
    std::size_t 偏移 = 位置_;
    char32_t 结果 = 0;
    for (int i = 0; i < n; i++) {
        if (偏移 >= 源码_.size()) return 0;
        unsigned char 首字节 = static_cast<unsigned char>(源码_[偏移]);
        int 长度 = 1;
        char32_t 码点 = 首字节;
        if (首字节 >= 0xF0) { 长度 = 4; 码点 = 首字节 & 0x07; }
        else if (首字节 >= 0xE0) { 长度 = 3; 码点 = 首字节 & 0x0F; }
        else if (首字节 >= 0xC0) { 长度 = 2; 码点 = 首字节 & 0x1F; }
        for (int j = 1; j < 长度 && 偏移 + j < 源码_.size(); j++) {
            码点 = (码点 << 6) | (static_cast<unsigned char>(源码_[偏移 + j]) & 0x3F);
        }
        偏移 += 长度;
        结果 = 码点;
    }
    return 结果;
}

// 是否到达源码末尾
bool 词法分析器::到达末尾() const {
    return 位置_ >= 源码_.size();
}

// 当前位置
SourceLocation 词法分析器::当前位置() const {
    return SourceLocation(文件名_, 行_, 列_);
}

// 跳过空白与注释（支持//单行、/* */块注释嵌套）
void 词法分析器::跳过空白与注释() {
    bool 继续 = true;
    while (继续) {
        继续 = false;
        // 跳过空白
        while (!到达末尾()) {
            char32_t c = 查看();
            if (c == '\n') {
                前进();
                行_++;
                列_ = 1;
            } else if (c == ' ' || c == '\t' || c == '\r') {
                前进();
            } else {
                break;
            }
        }
        // 单行注释
        if (查看() == '/' && 查看第N个(1) == '/') {
            前进(); 前进();
            while (!到达末尾() && 查看() != '\n') 前进();
            继续 = true;
        }
        // 块注释（支持嵌套）
        if (查看() == '/' && 查看第N个(1) == '*') {
            前进(); 前进();
            int 嵌套深度 = 1;
            while (!到达末尾() && 嵌套深度 > 0) {
                if (查看() == '/' && 查看第N个(1) == '*') { 前进(); 前进(); 嵌套深度++; }
                else if (查看() == '*' && 查看第N个(1) == '/') { 前进(); 前进(); 嵌套深度--; }
                else 前进();
            }
            继续 = true;
        }
    }
}

// 读取标识符或关键字（UTF-8中文/ASCII字母开头）
Token 词法分析器::读取标识符或关键字() {
    SourceLocation 位置 = 当前位置();
    std::string 文本;
    while (!到达末尾()) {
        char32_t c = 查看();
        // 标识符字符：字母/数字/下划线/中文（码点>=0x80）
        bool 是标识符字符 = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') ||
                           (c >= '0' && c <= '9') || c == '_' || c >= 0x80;
        if (!是标识符字符) break;
        // 收集当前字符的UTF-8字节
        if (回退标记_) {
            文本 += 回退字符_;  // 简化：ASCII回退场景
            前进();
        } else {
            std::size_t 起始 = 位置_;
            前进();
            文本.append(源码_, 起始, 位置_ - 起始);
        }
    }
    // 判断是否为关键字
    TokenType 类型;
    if (查找关键字(文本, 类型)) {
        return Token(类型, 文本, 位置);
    }
    return Token(TokenType::标识符, 文本, 位置);
}

// 主入口：分析源码返回Token流
std::vector<Token> 词法分析器::分析(const std::string& 源码, const std::string& 文件名) {
    源码_ = 源码;
    文件名_ = 文件名;
    位置_ = 0;
    行_ = 1;
    列_ = 1;
    回退标记_ = false;

    std::vector<Token> tokens;
    while (true) {
        跳过空白与注释();
        if (到达末尾()) {
            tokens.push_back(Token(TokenType::文件结束, "", 当前位置()));
            break;
        }
        char32_t c = 查看();
        // 标识符/关键字
        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '_' || c >= 0x80) {
            tokens.push_back(读取标识符或关键字());
            continue;
        }
        // 数字
        if (c >= '0' && c <= '9') {
            tokens.push_back(读取数字());
            continue;
        }
        // 字符串/字符/运算符由后续Task细化实现（此处为骨架，完整实现在本Task Step 5补齐）
        // 简化：未知字符报告诊断并跳过
        诊断_.报告(诊断::错误(文件名_, 行_, 列_, "暂不支持该词法单元"));
        前进();
    }
    return tokens;
}

// 读取数字（骨架：完整实现在Step 5补齐，此处支持十进制整数）
Token 词法分析器::读取数字() {
    SourceLocation 位置 = 当前位置();
    std::string 文本;
    while (!到达末尾()) {
        char32_t c = 查看();
        if (c >= '0' && c <= '9') {
            文本 += static_cast<char>(c);
            前进();
        } else {
            break;
        }
    }
    Token token(TokenType::整数字面量, 文本, 位置);
    token.设置整数值(std::stoll(文本));
    return token;
}

}

    return Token(TokenType::文件结束, "", 当前位置());
}
```

```cpp
// 语法分析器实现：递归下降 + Pratt表达式解析 + 两趟解析
#include "cn_compiler/parser/parser.hpp"

namespace cn_compiler {

// 当前Token
const Token& 语法分析器::当前() const {
    return tokens_[位置_];
}

// 前进一个Token
void 语法分析器::前进() {
    if (位置_ < tokens_.size() - 1) 位置_++;
}

// 检查当前是否为目标类型
bool 语法分析器::检查(TokenType 类型) const {
    return 当前().类型() == 类型;
}

// 匹配目标类型并前进
bool 语法分析器::匹配(TokenType 类型, const std::string& 预期描述) {
    if (检查(类型)) {
        前进();
        return true;
    }
    报告错误("预期 " + 预期描述 + "，实际为 '" + 当前().文本() + "'");
    return false;
}

// 报告语法错误
void 语法分析器::报告错误(const std::string& 消息) {
    诊断_.报告(诊断::错误(当前().位置(), 消息));
}

// 解析类型：关键字类型 或 标识符（结构体/枚举/类名）
std::string 语法分析器::解析类型() {
    // 类型关键字：整8~整128/正8~正128/浮32/浮64/整数/小数/布尔/字符/字符串/空类型
    // 或 标识符（自定义类型名）
    if (当前().是关键字() && 当前().文本().find("整") != std::string::npos ||
        当前().文本() == "整数" || 当前().文本() == "小数" ||
        当前().文本() == "布尔" || 当前().文本() == "字符" ||
        当前().文本() == "字符串" || 当前().文本() == "空类型" ||
        当前().类型() == TokenType::标识符) {
        std::string 类型名 = 当前().文本();
        前进();
        return 类型名;
    }
    报告错误("预期类型，实际为 '" + 当前().文本() + "'");
    return "";
}

// 解析参数：类型 名称（如 整32 a）
std::unique_ptr<参数> 语法分析器::解析参数() {
    auto 参数节点 = std::make_unique<参数>();
    参数节点->类型 = 解析类型();
    参数节点->名称 = 当前().文本();
    前进();
    return 参数节点;
}

// 解析函数声明：函数 名称(参数列表) -> 返回类型 { 函数体 }
std::unique_ptr<函数声明> 语法分析器::解析函数声明() {
    auto 函数 = std::make_unique<函数声明>();
    函数->位置 = 当前().位置();
    前进();  // 跳过"函数"关键字
    函数->名称 = 当前().文本();
    前进();
    if (!匹配(TokenType::括号开, "'('")) return 函数;
    // 参数列表（可空）
    if (!检查(TokenType::括号关)) {
        while (true) {
            函数->参数列表.push_back(解析参数());
            if (!检查(TokenType::逗号)) break;
            前进();
        }
    }
    匹配(TokenType::括号关, "')'");
    // 返回类型（-> 类型，可省略表示空类型）
    if (检查(TokenType::箭头)) {
        前进();
        函数->返回类型 = 解析类型();
    }
    // 函数体（可为空=原型声明）
    if (检查(TokenType::花括号开)) {
        函数->函数体 = 解析代码块();
    }
    return 函数;
}

// 解析代码块：{ 语句列表 }
std::unique_ptr<代码块> 语法分析器::解析代码块() {
    auto 块 = std::make_unique<代码块>();
    匹配(TokenType::花括号开, "'{'");
    while (!检查(TokenType::花括号关) && !检查(TokenType::文件结束)) {
        块->语句列表.push_back(解析语句());
    }
    匹配(TokenType::花括号关, "'}'");
    return 块;
}

// 解析变量声明：变量 名称 [= 初始值] / 类型 名称 [= 初始值] / 常量/静态
std::unique_ptr<语句> 语法分析器::解析变量声明(bool 是静态) {
    auto 声明 = std::make_unique<变量声明>();
    声明->是静态 = 是静态;
    // "变量"关键字：类型推断；"常量"关键字：常量（常量Token在词法层归属标识符，此处按文本判断）
    if (当前().文本() == "变量") {
        前进();
        声明->名称 = 当前().文本();
        前进();
    } else if (当前().文本() == "常量") {
        声明->是常量 = true;
        前进();
        声明->类型 = 解析类型();
        声明->名称 = 当前().文本();
        前进();
    } else {
        声明->类型 = 解析类型();
        声明->名称 = 当前().文本();
        前进();
    }
    // 初始值（= 表达式）
    if (检查(TokenType::赋值)) {
        前进();
        声明->初始值 = 解析表达式();
    }
    return 声明;
}

// 解析语句（骨架：完整分支在Step 6补齐）
std::unique_ptr<语句> 语法分析器::解析语句() {
    // 返回语句
    if (检查(TokenType::关键字返回)) {
        auto 返回 = std::make_unique<返回语句>();
        前进();
        if (!检查(TokenType::花括号关) && !检查(TokenType::分号) &&
            !检查(TokenType::文件结束)) {
            返回->值 = 解析表达式();
        }
        return 返回;
    }
    // 表达式语句
    auto 语句节点 = std::make_unique<表达式语句>(解析表达式());
    return 语句节点;
}

std::unique_ptr<表达式> 语法分析器::解析基本表达式() {
    // 字面量
    if (检查(TokenType::整数字面量) || 检查(TokenType::浮点字面量) ||
        检查(TokenType::字符串字面量) || 检查(TokenType::字符字面量) ||
        检查(TokenType::关键字真) || 检查(TokenType::关键字假) ||
        检查(TokenType::关键字无)) {
        auto 字面量 = std::make_unique<字面量表达式>(当前().文本());
        前进();
        return 字面量;
    }
    // 标识符 / 调用
    if (检查(TokenType::标识符)) {
        std::string 名称 = 当前().文本();
        前进();
        if (检查(TokenType::括号开)) {
            auto 调用 = std::make_unique<调用表达式>(名称);
            前进();
            while (!检查(TokenType::括号关)) {
                调用->参数列表.push_back(解析表达式());
                if (!检查(TokenType::逗号)) break;
                前进();
            }
            匹配(TokenType::括号关, "')'");
            return 调用;
        }
        return std::make_unique<标识符表达式>(名称);
    }
    报告错误("预期表达式，实际为 '" + 当前().文本() + "'");
    前进();
    return std::make_unique<字面量表达式>("0");
}
```

```cpp
// 符号表：作用域链管理，记录变量/函数/类型符号
#pragma once
#include <optional>
#include <string>
#include <unordered_map>
#include <vector>

namespace cn_compiler {

// 符号表：支持嵌套作用域，同一作用域内禁止重复声明
class 符号表 {
public:
    // 进入新作用域
    void 进入作用域() { 作用域栈_.emplace_back(); }
    // 退出当前作用域
    void 退出作用域() { if (!作用域栈_.empty()) 作用域栈_.pop_back(); }

    // 插入符号：同作用域重复返回false
    bool 插入(const std::string& 名称, const std::string& 类型) {
        if (作用域栈_.empty()) 进入作用域();
        auto& 当前 = 作用域栈_.back();
        if (当前.find(名称) != 当前.end()) return false;
        当前[名称] = 类型;
        return true;
    }

    // 查找符号：从内到外，返回类型
    std::optional<std::string> 查找(const std::string& 名称) const {
        for (auto 迭代 = 作用域栈_.rbegin(); 迭代 != 作用域栈_.rend(); ++迭代) {
            auto 找到 = 迭代->find(名称);
            if (找到 != 迭代->end()) return 找到->second;
        }
        return std::nullopt;
    }

private:
    std::vector<std::unordered_map<std::string, std::string>> 作用域栈_;  // 作用域链
};

} // namespace cn_compiler
```

```cpp
    // 判断源类型能否隐式转换到目标类型
    // 规则（规格书3.7）：字符↔整数、字符串↔字符*、空类型*↔任意指针、
    // 整数->整64（宽化）、浮32->浮64（宽化）
    static bool 可隐式转换(const std::string& 源, const std::string& 目标) {
        if (源 == 目标) return true;
        // 字符 ↔ 整数
        if ((源 == "字符" && 目标 == "整数") || (源 == "整数" && 目标 == "字符")) return true;
        // 字符串 ↔ 字符*
        if ((源 == "字符串" && 目标 == "字符*") || (源 == "字符*" && 目标 == "字符串")) return true;
        // 整数 -> 整64 宽化
        if ((源 == "整数" && 目标 == "整64") || (源 == "整32" && 目标 == "整64")) return true;
        // 浮32 -> 浮64 宽化
        if (源 == "浮32" && 目标 == "浮64") return true;
        // 空类型* -> 任意指针
        if (源 == "空类型*" && !目标.empty() && 目标.back() == '*') return true;
        return false;
    }

    // 判断是否为内置类型关键字
    static bool 是内置类型(const std::string& 名称) {
        static const std::unordered_set<std::string> 内置类型 = {
            "整数", "小数", "整8", "整16", "整32", "整64", "整128",
            "正8", "正16", "正32", "正64", "正128",
            "浮32", "浮64", "布尔", "字符", "字符串", "空类型",
        };
        return 内置类型.count(名称) > 0;
    }

    // 获取类型大小（字节）
    static int 大小(const std::string& 名称) {
        if (名称 == "整8" || 名称 == "正8" || 名称 == "布尔") return 1;
        if (名称 == "整16" || 名称 == "正16") return 2;
        if (名称 == "整32" || 名称 == "正32" || 名称 == "浮32") return 4;
        if (名称 == "整64" || 名称 == "正64" || 名称 == "浮64" || 名称 == "字符") return 8;
        if (名称 == "整128" || 名称 == "正128") return 16;
        if (名称 == "整数") return 4;   // 整32别名
        if (名称 == "小数") return 8;   // 浮64别名
        return 0;  // 空类型/自定义类型由语义分析器计算
    }
};

} // namespace cn_compiler
```

```cpp
// 语义分析器实现：符号收集 + 基础类型检查（完整类型推断在阶段二完善）
#include "cn_compiler/semantic/semantic.hpp"

namespace cn_compiler {

// 检查程序
bool 语义分析器::检查(std::unique_ptr<程序>& 程序) {
    符号表_.进入作用域();  // 全局作用域
    // 预置内置函数符号（打印行等运行时函数，阶段一先注册原型）
    符号表_.插入("打印行", "空类型");
    // 遍历顶层声明
    for (auto& 声明 : 程序->顶层声明) {
        检查顶层声明(*声明);
    }
    return 诊断_.是否成功();
}

// 检查顶层声明
void 语义分析器::检查顶层声明(AST节点& 声明) {
    if (声明.类型() == 节点类型::函数声明) {
        检查函数(static_cast<函数声明&>(声明));
    }
    // 结构体/枚举/类/导入：阶段二/三实现
}

// 检查函数
void 语义分析器::检查函数(函数声明& 函数) {
    // 注册函数符号（重载支持：阶段二用签名区分）
    if (!符号表_.插入(函数.名称, 函数.返回类型.empty() ? "空类型" : 函数.返回类型)) {
        诊断_.报告(诊断::错误(函数.位置, "重复声明函数 '" + 函数.名称 + "'"));
        return;
    }
    // 无函数体（原型声明）直接返回
    if (!函数.函数体) return;
    // 参数作用域
    符号表_.进入作用域();
    for (auto& 参数 : 函数.参数列表) {
        if (!符号表_.插入(参数->名称, 参数->类型)) {
            诊断_.报告(诊断::错误(参数->位置, "重复声明参数 '" + 参数->名称 + "'"));
        }
    }
    // 检查函数体语句
    for (auto& 语句 : 函数.函数体->语句列表) {
        检查语句(*语句);
    }
    符号表_.退出作用域();
}

// 检查语句（骨架：完整分支在阶段二补齐，阶段一覆盖返回/表达式/变量声明）
void 语义分析器::检查语句(语句& 语句节点) {
    switch (语句节点.类型()) {
        case 节点类型::返回语句: {
            auto& 返回 = static_cast<返回语句&>(语句节点);
            if (返回.值) 检查表达式(*返回.值);
            break;
        }
        case 节点类型::表达式语句: {
            auto& 表达式语句 = static_cast<cn_compiler::表达式语句&>(语句节点);
            检查表达式(*表达式语句.表达式);
            break;
        }
        case 节点类型::变量声明: {
            auto& 声明 = static_cast<变量声明&>(语句节点);
            std::string 类型 = 声明.类型;
            // 类型推断：无显式类型时从初始值推断
            if (类型.empty() && 声明.初始值) {
                类型 = 检查表达式(*声明.初始值);
            } else if (声明.初始值) {
                std::string 初始类型 = 检查表达式(*声明.初始值);
                if (!类型系统::可隐式转换(初始类型, 类型)) {
                    诊断_.报告(诊断::错误(声明.位置,
                        "无法将 '" + 初始类型 + "' 隐式转换为 '" + 类型 + "'"));
                }
            }
            if (!符号表_.插入(声明.名称, 类型)) {
                诊断_.报告(诊断::错误(声明.位置, "重复声明变量 '" + 声明.名称 + "'"));
            }
            break;
        }
        default:
            // 控制流语句：阶段二实现
            break;
    }
}

// 检查表达式（骨架：完整类型推断在阶段二完善）
std::string 语义分析器::检查表达式(表达式& 表达式节点) {
    switch (表达式节点.类型()) {
        case 节点类型::字面量表达式: {
            auto& 字面量 = static_cast<字面量表达式&>(表达式节点);
            // 按文本前缀推断：数字/字符串/布尔
            if (字面量.值 == "真" || 字面量.值 == "假") return "布尔";
            if (!字面量.值.empty() && (字面量.值[0] == '"' || 字面量.值[0] == '\'')) return "字符串";
            return "整数";
        }
        case 节点类型::标识符表达式: {
            auto& 标识符 = static_cast<标识符表达式&>(表达式节点);
            auto 类型 = 符号表_.查找(标识符.名称);
            if (!类型) {
                诊断_.报告(诊断::错误(标识符.位置, "未定义符号 '" + 标识符.名称 + "'"));
                return "未知";
            }
            return *类型;
        }
        case 节点类型::二元运算: {
            return 检查二元运算(static_cast<二元运算&>(表达式节点));
        }
        default:
            return "未知";
    }
}

// 检查二元运算
std::string 语义分析器::检查二元运算(二元运算& 运算) {
    std::string 左类型 = 检查表达式(*运算.左);
    std::string 右类型 = 检查表达式(*运算.右);
    // 算术运算要求整型/浮点，比较与逻辑返回布尔
    switch (运算.运算) {
        case 运算符::加: case 运算符::减:
        case 运算符::乘: case 运算符::除: case 运算符::取余:
            if (!类型系统::可隐式转换(右类型, 左类型)) {
                诊断_.报告(诊断::错误(运算.位置,
                    "运算符操作数类型不匹配: '" + 左类型 + "' 与 '" + 右类型 + "'"));
            }
            return 左类型;
        case 运算符::等号等号: case 运算符::不等号:
        case 运算符::小于: case 运算符::大于:
        case 运算符::小于等于: case 运算符::大于等于:
        case 运算符::逻辑与: case 运算符::逻辑或:
            return "布尔";
        default:
            return 左类型;
    }
}

} // namespace cn_compiler
```

```cpp
    操作码 操作码;                          // 操作码

    std::string 目标寄存器;                 // 目标虚拟寄存器（%vN）

// 基本块：CFG节点

    // 终止指令信息（跳转/条件跳转/返回）

    std::string 真目标;                     // 条件跳转真分支目标块名
    std::string 假目标;                     // 条件跳转假分支目标块名
    std::string 返回值;                     // 返回指令的返回值寄存器

// IR函数：函数签名 + 基本块列表 + Phi节点
struct IR函数 {

    std::string 返回类型;                   // 返回类型
```

```cpp
// IR生成器实现：AST降级为SSA形式IR（阶段一覆盖算术/调用/返回/常量）
#include "cn_compiler/ir/ir_generator.hpp"

namespace cn_compiler {

// 源码类型 -> IR类型映射（规格书7.5类型表示）
std::string IR生成器::类型映射(const std::string& 类型) {
    if (类型 == "整8") return "i8";
    if (类型 == "整16") return "i16";
    if (类型 == "整32" || 类型 == "整数") return "i32";
    if (类型 == "整64") return "i64";
    if (类型 == "整128") return "i128";
    if (类型 == "正8") return "u8";
    if (类型 == "正16") return "u16";
    if (类型 == "正32") return "u32";
    if (类型 == "正64") return "u64";
    if (类型 == "正128") return "u128";
    if (类型 == "浮32") return "f32";
    if (类型 == "浮64" || 类型 == "小数") return "f64";
    if (类型 == "布尔") return "i1";
    if (类型 == "字符" || 类型 == "字符串") return "ptr";
    if (类型 == "空类型") return "void";
    // 指针类型（类型* -> ptr）
    if (!类型.empty() && 类型.back() == '*') return "ptr";
    return "struct";  // 自定义类型（结构体/枚举/类）
}

// 新建虚拟寄存器
std::string IR生成器::新建虚拟寄存器() {
    return "%v" + std::to_string(虚拟寄存器计数_++);
}

// 添加指令
void IR生成器::添加指令(ir::基本块& 块, ir::操作码 操作码,
                         const std::vector<std::string>& 源操作数,
                         const std::string& 目标寄存器, const std::string& 类型,
                         const SourceLocation& 位置) {
    ir::指令 指令;
    指令.操作码 = 操作码;
    指令.源操作数 = 源操作数;
    指令.目标寄存器 = 目标寄存器;
    指令.类型 = 类型;
    指令.位置 = 位置;
    块.指令列表.push_back(指令);
}

// 生成函数
void IR生成器::生成函数(ir::IR模块& 模块, 函数声明& 函数) {
    ir::IR函数 ir函数;
    ir函数.名称 = 函数.名称;
    ir函数.返回类型 = 类型映射(函数.返回类型.empty() ? "空类型" : 函数.返回类型);
    // 参数列表
    int 参数编号 = 0;
    for (auto& 参数 : 函数.参数列表) {
        ir函数.参数列表.emplace_back(参数->名称, 类型映射(参数->类型));
        ir函数.参数编号映射[参数->名称] = 参数编号++;
    }
    // 入口基本块
    auto 入口块 = std::make_unique<ir::基本块>();
    入口块->名称 = "块0";
    // 函数体语句生成
    if (函数.函数体) {
        for (auto& 语句 : 函数.函数体->语句列表) {
            生成语句(模块, ir函数, *入口块, *语句);
        }
    }
    // 无返回语句时默认返回（空类型函数）
    if (ir函数.返回类型 == "void" && 入口块->终止类型.empty()) {
        入口块->终止类型 = "返回";
    }
    ir函数.基本块列表.push_back(std::move(入口块));
    模块.函数列表.push_back(std::move(ir函数));
}

// 生成语句（阶段一覆盖返回/表达式语句/变量声明）
void IR生成器::生成语句(ir::IR模块& 模块, ir::IR函数& 函数, ir::基本块& 当前块, 语句& 语句节点) {
    switch (语句节点.类型()) {
        case 节点类型::返回语句: {
            auto& 返回 = static_cast<返回语句&>(语句节点);
            if (返回.值) {
                当前块.返回值 = 生成表达式(模块, 函数, 当前块, *返回.值);
            }
            当前块.终止类型 = "返回";
            break;
        }
        case 节点类型::表达式语句: {
            auto& 表达式语句 = static_cast<cn_compiler::表达式语句&>(语句节点);
            生成表达式(模块, 函数, 当前块, *表达式语句.表达式);
            break;
        }
        case 节点类型::变量声明: {
            auto& 声明 = static_cast<变量声明&>(语句节点);
            if (声明.初始值) {
                // 变量声明生成常量定义指令（局部变量栈分配在阶段二完善）
                std::string 值寄存器 = 生成表达式(模块, 函数, 当前块, *声明.初始值);
                std::string 目标 = 新建虚拟寄存器();
                添加指令(当前块, ir::操作码::常量定义, {值寄存器}, 目标,
                         类型映射(声明.类型.empty() ? "整32" : 声明.类型), 声明.位置);
            }
            break;
        }
        default:
            // 控制流语句：阶段二实现
            break;
    }
}

// 生成表达式（阶段一覆盖字面量/标识符/二元运算/调用）
std::string IR生成器::生成表达式(ir::IR模块& 模块, ir::IR函数& 函数, ir::基本块& 当前块, 表达式& 表达式节点) {
    switch (表达式节点.类型()) {
        case 节点类型::字面量表达式: {
            auto& 字面量 = static_cast<字面量表达式&>(表达式节点);
            std::string 目标 = 新建虚拟寄存器();
            std::string 类型 = 类型映射("整32");  // 阶段一简化：数字按整32
            if (字面量.值 == "真" || 字面量.值 == "假") 类型 = "i1";
            添加指令(当前块, ir::操作码::常量定义, {字面量.值}, 目标, 类型, 字面量.位置);
            return 目标;
        }
        case 节点类型::标识符表达式: {
            auto& 标识符 = static_cast<标识符表达式&>(表达式节点);
            // 阶段一简化：参数直接引用虚拟寄存器（参数编号与虚拟寄存器编号对应）
            return "%v" + std::to_string(函数.参数编号映射[标识符.名称]);
        }
        case 节点类型::二元运算: {
            auto& 二元 = static_cast<二元运算&>(表达式节点);
            std::string 左 = 生成表达式(模块, 函数, 当前块, *二元.左);
            std::string 右 = 生成表达式(模块, 函数, 当前块, *二元.右);
            std::string 目标 = 新建虚拟寄存器();
            // 运算符 -> IR操作码映射（整型算术，浮点变体阶段二完善）
            ir::操作码 运算码;
            switch (二元.运算) {
                case 运算符::加: 运算码 = ir::操作码::加; break;
                case 运算符::减: 运算码 = ir::操作码::减; break;
                case 运算符::乘: 运算码 = ir::操作码::乘; break;
                case 运算符::除: 运算码 = ir::操作码::除; break;
                case 运算符::取余: 运算码 = ir::操作码::取余; break;
                default: 运算码 = ir::操作码::无操作; break;
            }
            添加指令(当前块, 运算码, {左, 右}, 目标, "i32", 二元.位置);
            return 目标;
        }
        case 节点类型::调用表达式: {
            auto& 调用 = static_cast<调用表达式&>(表达式节点);
            std::vector<std::string> 参数寄存器;
            for (auto& 参数 : 调用.参数列表) {
                参数寄存器.push_back(生成表达式(模块, 函数, 当前块, *参数));
            }
            std::string 目标 = 新建虚拟寄存器();
            添加指令(当前块, ir::操作码::直接调用, 参数寄存器, 目标, "i32", 调用.位置);
            // 记录被调用函数名（阶段一简化：指令源操作数首元素存函数名）
            当前块.指令列表.back().源操作数.insert(当前块.指令列表.back().源操作数.begin(), 调用.函数名);
            return 目标;
        }
        default:
            return "";
    }
}

// 主入口
ir::IR模块 IR生成器::生成(程序& 程序, 符号表& 符号表) {
    (void)符号表;  // 阶段一暂未使用符号表（参数编号映射简化）
    ir::IR模块 模块;
    虚拟寄存器计数_ = 0;
    for (auto& 声明 : 程序.顶层声明) {
        if (声明->类型() == 节点类型::函数声明) {
            生成函数(模块, static_cast<函数声明&>(*声明));
        }
    }
    return 模块;
}

} // namespace cn_compiler
```

```cpp
// Win x64代码生成器实现：IR指令映射为x86-64汇编
#include "cn_compiler/codegen/x64/x64_codegen.hpp"
#include <sstream>

namespace cn_compiler {

// 参数编号 -> Win x64寄存器（前4个整型参数）
std::string X64代码生成器::参数寄存器(int 编号) const {
    switch (编号) {
        case 0: return "rcx";
        case 1: return "rdx";
        case 2: return "r8";
        case 3: return "r9";
        default: return "rsp+" + std::to_string(40 + (编号 - 4) * 8);  // 栈上参数
    }
}

// 名字修饰：中文符号 UTF-8十六进制编码（规格书附录C）
std::string X64代码生成器::名字修饰(const std::string& 名称) {
    // 简化实现：? + 十六进制编码（阶段一ASCII符号直接?名@@Y...@Z）
    std::stringstream 流;
    流 << "?";
    for (unsigned char c : 名称) {
        char 缓冲[8];
        snprintf(缓冲, sizeof(缓冲), "%02X", c);
        流 << 缓冲;
    }
    流 << "@@Y";
    return 流.str();
}

// 指令选择：IR操作码 -> x86-64助记符
std::string X64代码生成器::指令选择(ir::操作码 操作码, const std::string& 类型) const {
    // 32位整数运算使用 eax/ebx 等寄存器，浮点使用 xmm（阶段二完善）
    switch (操作码) {
        case ir::操作码::加: return "add";
        case ir::操作码::减: return "sub";
        case ir::操作码::乘: return "imul";
        case ir::操作码::除: return "idiv";
        case ir::操作码::取余: return "idiv";  // 余数在rdx，需额外处理
        default: return "; 未映射指令";
    }
}

// 生成单个函数汇编
std::string X64代码生成器::生成函数汇编(const ir::IR函数& 函数) {
    std::stringstream 汇编;
    // 函数声明（PROC/ENDP，规格书8.8示例）
    汇编 << "?函数@" << 名字修饰(函数.名称) << "@@Y" << " PROC\n";
    // prologue：保存rbp，建立栈帧
    汇编 << "    push    rbp\n";
    汇编 << "    mov     rbp, rsp\n";
    // 参数映射：前4个参数从寄存器读取到虚拟寄存器对应栈槽
    // （阶段一简化：参数直接使用寄存器，虚拟寄存器与参数编号一致）
    // 指令选择：遍历每个基本块
    for (auto& 块 : 函数.基本块列表) {
        汇编 << 块->名称 << ":\n";
        for (auto& 指令 : 块->指令列表) {
            std::string 助记符 = 指令选择(指令.操作码, 指令.类型);
            if (助记符[0] == ';') {
                汇编 << "    " << 助记符 << " ; " << 指令.目标寄存器 << " = "
                      << (指令.源操作数.empty() ? "" : 指令.源操作数[0]) << "\n";
                continue;
            }
            // 阶段一简化：目标/源操作数使用参数寄存器（rax/rcx/rdx）
            // 完整寄存器分配在阶段四实现
            if (!指令.源操作数.empty()) {
                汇编 << "    mov     eax, " << 参数寄存器(0) << "\n";
                汇编 << "    " << 助记符 << "     eax, " << 参数寄存器(1) << "\n";
            }
        }
        // 终止指令：返回
        if (块->终止类型 == "返回") {
            汇编 << "    mov     eax, eax\n";  // 阶段一：返回值在eax
        }
    }
    // epilogue
    汇编 << "    pop     rbp\n";
    汇编 << "    ret\n";
    汇编 << "ENDP\n";
    return 汇编.str();
}

// 主入口：生成完整汇编文件
std::string X64代码生成器::生成汇编(const ir::IR模块& 模块) {
    std::stringstream 汇编;
    汇编 << "; CN语言编译器生成的Win x64汇编\n";
    汇编 << "; 目标平台: win-x64\n";
    汇编 << ".code\n";
    for (const auto& 函数 : 模块.函数列表) {
        汇编 << 生成函数汇编(函数);
    }
    汇编 << "END\n";
    return 汇编.str();
}

} // namespace cn_compiler
```

> **实现说明：** 阶段一代码生成器采用最简策略（参数直通寄存器、返回值eax），验证IR→汇编链路贯通；完整寄存器分配、栈帧布局、溢出处理在阶段四（规格书8.3/8.4）实现。
